25// See: https://en.wikipedia.org/wiki/Master_boot_record#PTE
26typedef struct __attribute__((packed)) _mbr_partition_entry
27{
28 /* 0x00 */ u8 status; // Status or physical drive (bit 7 set is for active or bootable, old MBRs only accept 0x80, 0x00 means inactive, and 0x01–0x7F stand for invalid)
29 /* 0x01 */ u8 chs_first_lba[3]; // CHS address of first absolute sector in partition
30 /* 0x04 */ u8 partition_type; // Type of partition, see: https://en.wikipedia.org/wiki/Partition_type
31 /* 0x05 */ u8 chs_last_lba[3]; // CHS address of last absolute sector in partition
32 /* 0x08 */ u32 first_lba; // LBA of first absolute sector in the partition
33 /* 0x0C */ u32 sector_count; // Number of sectors in partition
34} mbr_partition_entry;

37typedef struct __attribute__((packed)) _master_boot_record
38{
39 /* 0x00 */ u8 bootstrap_code1[218]; // Bootstrap code area (part 1)

41 /* 0xE0 */ u8 bootstrap_code2[216]; // Bootstrap code area (part 2, code entry at 0x0000)
42 /* 0x1B8 */ u32 disk_signature; // Disk signature (optional; UEFI, Linux, Windows NT family and other OSes)
43 /* 0x1BC */ u16 copy_protection; // 0x0000 (0x5A5A if copy-protected)
44 /* 0x1BE */ mbr_partition_entry primary_partitions[4]; // Partition table (for primary partitions)
