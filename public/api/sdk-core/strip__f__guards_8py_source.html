strip_f_guards.py
1#!/usr/bin/env python3
2"""Strip F_<name> compile-time guard pairs from a ps2sdk kernel source
3file, leaving the body unconditional and leaving every OTHER conditional
4(__OPTIMIZE_SIZE__, USE_KMODE, etc.) completely untouched - including
5ones nested INSIDE an F_ block, which lose their enclosing guard but keep
6their own.
7
8Recognizes several guard spellings, all equally real - found the hard
9way, not up front: an initial pass over kernel.S/sifcmd.c/etc only
10handled #1 below, and go test/the golden-file suite had no way to notice
11#2/#3 being silently left as real (always-false, since no F_ macro is
12ever -D'd anymore) conditionals - loadfile.c's ~14 functions
13(SifLoadModule and friends) got compiled to nothing, zero errors, only
14found by grepping every already-stripped file for leftover "F_" text and
15checking each hit by hand. #4 was found the exact same way, in a
16completely different codebase (gsKit, not ps2sdk):
17 1. #ifdef F_<name> - kernel.S, most files
18 2. #if defined(F_<name>) - loadfile.c, exclusively
19 3. #if defined(F_A) || defined(F_B) [|| ...] - sjis.c, guards shared
20 data used by multiple F_-gated functions (e.g. a lookup table both
21 isSpecialSJIS and isSpecialASCII need) - once every one of those F_
22 names' own guards are gone, the shared data must always be kept too,
23 so this form strips exactly like #1/#2. A #if mixing an F_ name with
24 any NON-F_ condition is a real structural case this pattern doesn't
25 predict and is left alone, matching the #else/#elif refusal below.
26 4. #if F_<name> (bare, no ifdef/defined()) - gsKit's ee/gs/*.c
27 almost exclusively use this spelling instead of #1/#2 - works because
28 an undefined identifier in a preprocessor #if evaluates to 0, making
29 it functionally identical to #ifdef whenever nothing ever defines the
30 symbol to anything but 0/1. Also has a compound form (bare terms
31 joined by ||), handled the same all-F_-only-or-refuse way as #3.
32
33Stack-based: every #if/#ifdef/#ifndef pushes a marker noting whether it's
34an F_ guard (any of the 3 forms above) or not. Every #endif pops the
35matching marker.
36
37A bare F_ guard (#ifdef F_X ... #endif, no #else) has its two guard lines
38dropped, body kept. A small number of F_ guards DO have a real #else -
39inspected by hand (osd_config.c: converttobcd/convertfrombcd/
40__adjustTime) and confirmed to be the same recurring shape every time:
41the #ifdef branch is the real definition, the #else branch is JUST an
42`extern` forward-declaration stub, needed only so some OTHER F_-gated
43function in the same file compiling in isolation still has a prototype
44for this one. Once every function compiles together in one pass, that
45stub is redundant - the real definition from its own #ifdef branch
46already provides both the declaration and the implementation to
47everything below it in the same file. So for an F_ guard WITH an #else:
48keep the #ifdef branch's body, drop the #else branch and its content
49entirely. An #elif on an F_ guard has never been observed and stays a
50hard refusal - a real structural surprise this pattern doesn't predict,
51worth a human look rather than a guess.
52"""
53import re
54import sys
55
56_IDENT = r"[A-Za-z_][A-Za-z0-9_]*"
57_IFDEF_F = re.compile(rf"^#ifdef\s+(F_{_IDENT})$")
58_IF_DEFINED_ONE = re.compile(rf"^#if\s+defined\s*\‍(\s*(F_{_IDENT})\s*\‍)$")
59_IF_BARE_ONE = re.compile(rf"^#if\s+(F_{_IDENT})$")
60_DEFINED_TERM = re.compile(rf"defined\s*\‍(\s*(F_{_IDENT})\s*\‍)")
61_BARE_TERM = re.compile(rf"(F_{_IDENT})")
62
63def _is_f_guard(stripped_line):
64 """Returns True if stripped_line is any of the 5 recognized F_ guard
65 forms. A compound form requires EVERY "||" operand to name an F_
66 symbol (either bare or defined(...)) - a mixed condition (an F_ name
67 alongside something else) does not match, by design.
68
69 Form 4, bare "#if F_<name>" (no ifdef, no defined()), was found in
70 gsKit (ee/gs/src/*.c almost exclusively use this spelling, not the
71 ps2sdk-style #ifdef/defined() forms) - it works because an undefined
72 identifier in a preprocessor #if expression evaluates to 0, so
73 "#if F_gsKit_vram_alloc" is functionally identical to
74 "#ifdef F_gsKit_vram_alloc" whenever nothing ever defines it as
75 anything other than 0/1. A 4th real-world spelling of the exact same
76 ps2dev-community compile-once-per-function convention, found the same
77 way the first 3 were: grepping leftover F_ text after a first strip
78 pass looked clean by the OLD recognized-forms list."""
79 if _IFDEF_F.match(stripped_line):
80 return True
81 if _IF_DEFINED_ONE.match(stripped_line):
82 return True
83 if _IF_BARE_ONE.match(stripped_line):
84 return True
85 if stripped_line.startswith("#if ") and "||" in stripped_line:
86 cond = stripped_line[len("#if "):].strip()
87 terms = [t.strip() for t in cond.split("||")]
88 matched = [
89 _DEFINED_TERM.fullmatch(t) is not None or _BARE_TERM.fullmatch(t) is not None
90 for t in terms
91 ]
92 if terms and all(matched):
93 return True
94 return False

96def strip(path):
97 # latin-1, not utf-8: byte-transparent (maps 0-255 -> code points
98 # 0-255 1:1) so a file containing non-UTF-8 bytes in string literals
99 # or comments (sjis.c legitimately does - it's Shift-JIS text
100 # handling code) round-trips unchanged. Safe here because every
101 # actual decision this script makes is ASCII-level line-prefix
102 # matching (#ifdef/#endif/#else), never touching multi-byte content.
103 #
104 # newline="": disables Python's universal-newline translation on
105 # both read and write, so each line keeps its ORIGINAL terminator
106 # (line.strip() below still works - it strips \r and/or \n either
107 # way). Without this, running on Windows silently rewrites every
108 # line ending to \r\n regardless of the source file's own convention
109 # - found for real vendoring gsKit (LF-only upstream), where a first
110 # pass turned the whole file into a bogus wall-of-CRLF diff against
111 # nothing but line-ending noise, even though every actual guard-strip
112 # decision was correct.
113 with open(path, "r", encoding="latin-1", newline="") as f:
114 lines = f.readlines()
115
116 stack = [] # each entry: True if this level is an F_ guard
117 skipping_else = False # true while dropping an F_ guard's #else branch
118 out = []
119 removed = 0
120 for i, line in enumerate(lines, 1):
121 stripped = line.strip()
122 if stripped.startswith("#if") and _is_f_guard(stripped):
123 stack.append(True)
124 removed += 1
125 continue # drop this line
126 if stripped.startswith("#if"):
127 stack.append(False)
128 if not skipping_else:
129 out.append(line)
130 continue
131 if stripped.startswith("#elif"):
132 if stack and stack[-1] is True:
133 raise SystemExit(f"{path}:{i}: F_ guard has an #elif - refusing, needs a human")
134 if not skipping_else:
135 out.append(line)
136 continue
137 if stripped.startswith("#else"):
138 if stack and stack[-1] is True:
139 skipping_else = True
140 continue # drop this line, and everything until its #endif
141 if not skipping_else:
142 out.append(line)
143 continue
144 if stripped.startswith("#endif"):
145 if not stack:
146 raise SystemExit(f"{path}:{i}: unmatched #endif")
147 was_f_guard = stack.pop()
148 if was_f_guard:
149 skipping_else = False
150 continue # drop this line
151 if not skipping_else:
152 out.append(line)
153 continue
154 if not skipping_else:
155 out.append(line)
156
157 if stack:
158 raise SystemExit(f"{path}: unclosed conditional(s) at EOF: {stack}")
159
160 with open(path, "w", encoding="latin-1", newline="") as f:
161 f.writelines(out)
162 print(f"{path}: removed {removed} F_ guard pairs")
163
164if __name__ == "__main__":
165 for p in sys.argv[1:]:
166 strip(p)
