60int __fileXioOpenHelper(_libcglue_fdman_fd_info_t *info, const char *buf, int flags, mode_t mode)
61{
62 int iop_flags = 0;
63 int is_dir = 0;
64 int iop_fd;
65
66 // newlib flags differ from iop flags
67 if ((flags & 3) == O_RDONLY) iop_flags |= IOP_O_RDONLY;
68 if ((flags & 3) == O_WRONLY) iop_flags |= IOP_O_WRONLY;
69 if ((flags & 3) == O_RDWR ) iop_flags |= IOP_O_RDWR;
70 if (flags & O_NONBLOCK) iop_flags |= IOP_O_NBLOCK;
71 if (flags & O_APPEND) iop_flags |= IOP_O_APPEND;
72 if (flags & O_CREAT) iop_flags |= IOP_O_CREAT;
73 if (flags & O_TRUNC) iop_flags |= IOP_O_TRUNC;
74 if (flags & O_EXCL) iop_flags |= IOP_O_EXCL;
75 //if (flags & O_???) iop_flags |= IOP_O_NOWAIT;
76 if (flags & O_DIRECTORY) {
77 iop_flags |= IOP_O_DIROPEN;
78 is_dir = 1;
79 }
80
81 iop_fd = is_dir ? fileXioDopen(buf) : fileXioOpen(buf, iop_flags, mode);
82 if (iop_fd >= 0) {
83 _fxio_file_info_t *userdata;
84 int buf_len;
85
86 buf_len = strlen(buf);
87 userdata = malloc(sizeof(_fxio_file_info_t) + buf_len + 1);
88 if (userdata == NULL)
89 {
90 return -ENOMEM;
91 }
92 userdata->fd = iop_fd;
93 memcpy(userdata->filename, buf, buf_len);
94 userdata->filename[buf_len] = '\x00';
95 info->userdata = (void *)userdata;
96 info->ops = is_dir ? &__fileXio_fdman_ops_dir : &__fileXio_fdman_ops_file;
97 return 0;
98 }
99 return iop_fd;
100}

128static time_t io_to_posix_time(const unsigned char *ps2time)
129{
130 struct tm tim;
131 tim.tm_sec = ps2time[1];
132 tim.tm_min = ps2time[2];
133 tim.tm_hour = ps2time[3];
134 tim.tm_mday = ps2time[4];
135 tim.tm_mon = ps2time[5] - 1;
136 tim.tm_year = ((u16)ps2time[6] | ((u16)ps2time[7] << 8)) - 1900;
137 return mktime(&tim);
138}

140static mode_t iox_to_posix_mode(unsigned int ps2mode)
141{
142 mode_t posixmode = 0;
143 if (ps2mode & FIO_S_IFREG) posixmode |= S_IFREG;
144 if (ps2mode & FIO_S_IFDIR) posixmode |= S_IFDIR;
145 if (ps2mode & FIO_S_IRUSR) posixmode |= S_IRUSR;
146 if (ps2mode & FIO_S_IWUSR) posixmode |= S_IWUSR;
147 if (ps2mode & FIO_S_IXUSR) posixmode |= S_IXUSR;
148 if (ps2mode & FIO_S_IRGRP) posixmode |= S_IRGRP;
149 if (ps2mode & FIO_S_IWGRP) posixmode |= S_IWGRP;
150 if (ps2mode & FIO_S_IXGRP) posixmode |= S_IXGRP;
151 if (ps2mode & FIO_S_IROTH) posixmode |= S_IROTH;
152 if (ps2mode & FIO_S_IWOTH) posixmode |= S_IWOTH;
153 if (ps2mode & FIO_S_IXOTH) posixmode |= S_IXOTH;
154 return posixmode;
155}

157static void fill_stat(struct stat *stat, const iox_stat_t *fiostat)
158{
159 stat->st_dev = 0;
160 stat->st_ino = 0;
161 stat->st_mode = iox_to_posix_mode(fiostat->mode);
162 stat->st_nlink = 0;
163 stat->st_uid = 0;
164 stat->st_gid = 0;
165 stat->st_rdev = 0;
166 stat->st_size = ((off_t)fiostat->hisize << 32) | (off_t)fiostat->size;
167 stat->st_atime = io_to_posix_time(fiostat->atime);
168 stat->st_mtime = io_to_posix_time(fiostat->mtime);
169 stat->st_ctime = io_to_posix_time(fiostat->ctime);
170 stat->st_blksize = 16*1024;
171 stat->st_blocks = stat->st_size / 512;
172}

174int __fileXioGetstatHelper(const char *path, struct stat *buf) {
175 iox_stat_t fiostat;
176
177 if (strncmp(path, "tty", 3) == 0 && path[3] >= '0' && path[3] <= '9' && path[4] == ':')
178 {
179 memset(buf, 0, sizeof(struct stat));
180 buf->st_mode = S_IFCHR;
181 return 0;
182 }
183
184 if (fileXioGetStat(path, &fiostat) < 0) {
185 errno = ENOENT;
186 return -1;
187 }
188
189 fill_stat(buf, &fiostat);
190
191 return 0;
192}

314int __fileXioDreadHelper(void *userdata, struct dirent *dir)
315{
316 int rv;
317 int fd;
318 iox_dirent_t ioxdir;
319
320 // Took from iox_dirent_t
321 #define __MAXNAMLEN 256
322
323 fd = __fileXioGetFdHelper(userdata);
324 if (fd < 0)
325 {
326 return fd;
327 }
328
329 rv = fileXioDread(fd, &ioxdir);
330 if (rv < 0) {
331 return -ENOENT;
332 }
333
334 dir->d_fileno = rv; // TODO: This number should be in theory a unique number per file
335 snprintf(dir->d_name, sizeof(dir->d_name), "%.*s", (int)(sizeof(ioxdir.name) - 1), ioxdir.name);
336 dir->d_reclen = 0;
337 switch (ioxdir.stat.mode & FIO_S_IFMT) {
338 case FIO_S_IFLNK: dir->d_type = DT_LNK; break;
339 case FIO_S_IFDIR: dir->d_type = DT_DIR; break;
340 case FIO_S_IFREG: dir->d_type = DT_REG; break;
341 default: dir->d_type = DT_UNKNOWN; break;
342 }
343
344 return rv;
345}

347int __fileXioLseekDirHelper(void *userdata, int offset, int whence)
348{
349 int i;
350 int fd;
351 char *filename;
352 _fxio_file_info_t *finfo;
353 int uid;
354 struct dirent dir;
355
356 fd = __fileXioGetFdHelper(userdata);
357 if (fd < 0)
358 {
359 return fd;
360 }
361
362 filename = __fileXioGetFilenameHelper(userdata);
363
364 if (filename == NULL)
365 {
366 return -EINVAL;
367 }
368
369 if (userdata == NULL)
370 {
371 return -EINVAL;
372 }
373
374 finfo = (_fxio_file_info_t *)userdata;
375
376 if (whence != SEEK_SET)
377 {
378 return -EINVAL;
379 }
380
381 fileXioDclose(fd);
382 uid = fileXioDopen(filename);
383 finfo->fd = uid;
384 for (i = 0; i < offset; i++) {
385 __fileXioDreadHelper(userdata, &dir);
386 }
387
388 return offset;
389}
