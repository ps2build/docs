123s32 sceCdInit(s32 mode)
124{
125 if (_CdSyncS(1))
126 return 0;
127 sceSifInitRpc(0);
128 memset(&clientInit, 0, sizeof(clientInit));
129
130 while (1) {
131 if (sceSifBindRpc(&clientInit, CD_SERVER_INIT, 0) < 0) {
132 if (CdDebug > 0)
133 printf("Libcdvd bind err CD_Init\n");
134 } else if (clientInit.server != 0)
135 break;
136
137 nopdelay();
138 }
139
140 initMode = mode;
141 if (sceSifCallRpc(&clientInit, 0, 0, &initMode, sizeof(initMode), &cdInitRecvBuff, sizeof(cdInitRecvBuff), 0, 0) < 0)
142 return 0;
143 initVersionCdvdfsv = cdInitRecvBuff.m_cdvdfsv_version;
144 initVersionCdvdman = cdInitRecvBuff.m_cdvdman_version;
145 if (mode == SCECdEXIT) {
146 if (CdDebug > 0)
147 printf("Libcdvd Exit\n");
148 _CdSemaExit();
149 nCmdSemaId = 0;
150 sCmdSemaId = 0;
151 callbackSemaId = 0;
152 } else {
153 _CdSemaInit();
154 }
155
156 return 1;
157}

172s32 sceCdSearchFile(sceCdlFILE *file, const char *name)
173{
174 _CdSemaInit();
175 if (PollSema(nCmdSemaId) != nCmdSemaId)
176 return 0;
177 nCmdNum = CD_SERVER_SEARCHFILE;
178 if (sceCdSync(1)) {
179 SignalSema(nCmdSemaId);
180 return 0;
181 }
182 sceSifInitRpc(0);
183 if (HasIopRebootedSinceLastCall())
184 memset(&clientSearchFile, 0, sizeof(clientSearchFile));
185 if (!clientSearchFile.server) {
186 while (1) {
187 if (sceSifBindRpc(&clientSearchFile, CD_SERVER_SEARCHFILE, 0) < 0) {
188 if (CdDebug > 0)
189 printf("libsceCdvd bind err sceCdSearchFile\n");
190 }
191 if (clientSearchFile.server != 0)
192 break;
193
194 nopdelay();
195 }
196 }
197
198 strlcpy(searchFileSendBuff.name, name, sizeof(searchFileSendBuff.name));
199 searchFileSendBuff.dest = &searchFileSendBuff;
200
201 if (CdDebug > 0)
202 printf("ee call cmd search %s\n", searchFileSendBuff.name);
203 if (sceSifCallRpc(&clientSearchFile, 0, 0, &searchFileSendBuff, sizeof(SearchFilePkt), nCmdRecvBuff, 4, NULL, NULL) < 0) {
204 SignalSema(nCmdSemaId);

206 }

226s32 sceCdDiskReady(s32 mode)
227{
228 if (CdDebug > 0)
229 printf("DiskReady 0\n");
230
231 _CdSemaInit();
232 if (PollSema(sCmdSemaId) != sCmdSemaId)
233 return SCECdNotReady;
234 if (_CdSyncS(1)) {
235 SignalSema(sCmdSemaId);
236 return SCECdNotReady;
237 }
238
239 sceSifInitRpc(0);
240 if (HasIopRebootedSinceLastCall())
241 memset(&clientDiskReady, 0, sizeof(clientDiskReady));
242 if (!clientDiskReady.server) {
243 while (1) {
244 if (sceSifBindRpc(&clientDiskReady, CD_SERVER_DISKREADY, 0) < 0) {
245 if (CdDebug > 0)
246 printf("Libcdvd bind err CdDiskReady\n");
247 }
248 if (clientDiskReady.server != 0)
249 break;
250
251 nopdelay();
252 }
253 }
254 diskReadyMode = mode;
255
256 if (sceSifCallRpc(&clientDiskReady, 0, 0, &diskReadyMode, 4, sCmdRecvBuff, 4, NULL, NULL) < 0) {
257 SignalSema(sCmdSemaId);
258 return 6;
259 }
260 if (CdDebug > 0)
261 printf("DiskReady ended\n");
262
263 SignalSema(sCmdSemaId);
264 return *(s32 *)UNCACHED_SEG(sCmdRecvBuff);

267void _CdSemaInit(void)
268{
269 struct t_ee_sema semaParam;
270
271 // return if both semaphores are already inited
272 if (nCmdSemaId != 0 && sCmdSemaId != 0)
273 return;
274
275 semaParam.init_count = 1;
276 semaParam.max_count = 1;
277 semaParam.option = 0;
278 nCmdSemaId = CreateSema(&semaParam);
279 if (nCmdSemaId < 0)
280 nCmdSemaId = 0;
281 sCmdSemaId = CreateSema(&semaParam);
282 if (sCmdSemaId < 0)
283 sCmdSemaId = 0;
284
285 semaParam.init_count = 0;
286 callbackSemaId = CreateSema(&semaParam);
287 if (callbackSemaId < 0)
288 callbackSemaId = 0;
289
290 cbSema = 0;
291}

305s32 sceCdInitEeCB(s32 priority, void *stackAddr, s32 stackSize)
306{
307 // if callback thread has already been initialised, just change its priority
308 if (callbackThreadId != 0) {
309 ChangeThreadPriority(callbackThreadId, priority);
310 return 0;
311 }
312 // initialise callback thread
313 callbackThreadParam.stack_size = stackSize;
314 callbackThreadParam.gp_reg = &_gp;
315 callbackThreadParam.func = &_CdCallbackLoop;
316 callbackThreadParam.stack = stackAddr;
317 callbackThreadParam.initial_priority = priority;
318 callbackThreadId = CreateThread(&callbackThreadParam);
319 StartThread(callbackThreadId, NULL);
320
321 return 1;
322}

343static void _CdCallbackLoop(void)
344{
345 while (1) {
346 WaitSema(callbackSemaId);
347
348 // if callback number if -1, stop callbck thread loop
349 if (CdCallbackNum == -1)
350 ExitThread();
351
352 if (CdDebug > 0)
353 printf("sceCdCallbackFunc = %08X CdCallbackNum = %d\n", (u32)sceCdCallbackFunc, CdCallbackNum);
354
355 // if callback function number and 'custom callback function' pointer are valid, do callback
356 if (sceCdCallbackFunc && CdCallbackNum)
357 sceCdCallbackFunc(CdCallbackNum);
358
359 cbSema = 0;
360 }
361}

364void _CdGenericCallbackFunction(void *funcNum)
365{
366 // set the currently executing function num
367 CdCallbackNum = *(s32 *)funcNum;
368 iSignalSema(nCmdSemaId);
369
370 // check if user callback is registered

372 if (sceCdCallbackFunc) {
373 iSignalSema(callbackSemaId);

377
378 // clear the currently executing function num
