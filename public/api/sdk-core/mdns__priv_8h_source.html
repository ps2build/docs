185/* MDNS states */
186typedef enum {

188 MDNS_STATE_OFF,
189 /* Waiting before probing can be started */
190 MDNS_STATE_PROBE_WAIT,
191 /* Probing the unique records */
192 MDNS_STATE_PROBING,
193 /* Waiting before announcing the probed unique records */
194 MDNS_STATE_ANNOUNCE_WAIT,
195 /* Announcing all records */
196 MDNS_STATE_ANNOUNCING,
