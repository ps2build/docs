75/*------------------------------------------------------------------------*/
76/* Create a Mutex */
77/*------------------------------------------------------------------------*/
78/* This function is called in f_mount function to create a new mutex
79/ or semaphore for the volume. When a 0 is returned, the f_mount function
80/ fails with FR_INT_ERR.
81*/
82
83int ff_mutex_create ( /* Returns 1:Function succeeded or 0:Could not create the mutex */
84 int vol /* Mutex ID: Volume mutex (0 to FF_VOLUMES - 1) or system mutex (FF_VOLUMES) */
85)
86{
87#if OS_TYPE == 0 /* Win32 */
88 Mutex[vol] = CreateMutex(NULL, FALSE, NULL);
89 return (int)(Mutex[vol] != INVALID_HANDLE_VALUE);
90
91#elif OS_TYPE == 1 /* uITRON */
92 T_CMTX cmtx = {TA_TPRI,1};
93
94 Mutex[vol] = acre_mtx(&cmtx);
95 return (int)(Mutex[vol] > 0);
96
97#elif OS_TYPE == 2 /* uC/OS-II */
98 OS_ERR err;
99
100 Mutex[vol] = OSMutexCreate(0, &err);
101 return (int)(err == OS_NO_ERR);
102
103#elif OS_TYPE == 3 /* FreeRTOS */
104 Mutex[vol] = xSemaphoreCreateMutex();
105 return (int)(Mutex[vol] != NULL);
106
107#elif OS_TYPE == 4 /* CMSIS-RTOS */
108 osMutexDef(cmsis_os_mutex);
109
110 Mutex[vol] = osMutexCreate(osMutex(cmsis_os_mutex));
111 return (int)(Mutex[vol] != NULL);
112
113#elif OS_TYPE == 5 /* IOP-PS2 */
114 iop_sema_t sp;
115
116 sp.initial = 1;
117 sp.max = 1;
118 sp.option = 0;
119 sp.attr = 0;
120 Mutex[vol] = CreateSema(&sp);
121 return (int)(Mutex[vol] >= 0);
122#endif
123}

128/*------------------------------------------------------------------------*/

146 OSMutexDel(Mutex[vol], OS_DEL_ALWAYS, &err);

151#elif OS_TYPE == 4 /* CMSIS-RTOS */
152 osMutexDelete(Mutex[vol]);
153
154#elif OS_TYPE == 5 /* IOP-PS2 */

156
157#endif
158}

161/*------------------------------------------------------------------------*/
162/* Request a Grant to Access the Volume */
163/*------------------------------------------------------------------------*/
164/* This function is called on enter file functions to lock the volume.
165/ When a 0 is returned, the file function fails with FR_TIMEOUT.
166*/
167
168int ff_mutex_take ( /* Returns 1:Succeeded or 0:Timeout */
169 int vol /* Mutex ID: Volume mutex (0 to FF_VOLUMES - 1) or system mutex (FF_VOLUMES) */
170)
171{
172#if OS_TYPE == 0 /* Win32 */
173 return (int)(WaitForSingleObject(Mutex[vol], FF_FS_TIMEOUT) == WAIT_OBJECT_0);
174
175#elif OS_TYPE == 1 /* uITRON */
176 return (int)(tloc_mtx(Mutex[vol], FF_FS_TIMEOUT) == E_OK);
177
178#elif OS_TYPE == 2 /* uC/OS-II */
179 OS_ERR err;
180
181 OSMutexPend(Mutex[vol], FF_FS_TIMEOUT, &err));
182 return (int)(err == OS_NO_ERR);
183
184#elif OS_TYPE == 3 /* FreeRTOS */
185 return (int)(xSemaphoreTake(Mutex[vol], FF_FS_TIMEOUT) == pdTRUE);
186
187#elif OS_TYPE == 4 /* CMSIS-RTOS */
188 return (int)(osMutexWait(Mutex[vol], FF_FS_TIMEOUT) == osOK);
189
190#elif OS_TYPE == 5 /* IOP-PS2 */
191 return (int)(WaitSema(Mutex[vol]) == 0);
192
193#endif
194}
