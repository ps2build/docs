10/*
11 * SETUP: Make sure we define everything we will need.
12 *
13 * We have create three types of pools:
14 * 1) MEMPOOL - standard pools
15 * 2) MALLOC_MEMPOOL - to be used by mem_malloc in mem.c
16 * 3) PBUF_MEMPOOL - a mempool of pbuf's, so include space for the pbuf struct
17 *
18 * If the include'r doesn't require any special treatment of each of the types
19 * above, then will declare #2 & #3 to be just standard mempools.
20 */
21#ifndef LWIP_MALLOC_MEMPOOL
22/* This treats "malloc pools" just like any other pool.
23 The pools are a little bigger to provide 'size' as the amount of user data. */
24#define LWIP_MALLOC_MEMPOOL(num, size) LWIP_MEMPOOL(POOL_##size, num, (size + LWIP_MEM_ALIGN_SIZE(sizeof(struct memp_malloc_helper))), "MALLOC_"#size)
25#define LWIP_MALLOC_MEMPOOL_START
26#define LWIP_MALLOC_MEMPOOL_END
27#endif /* LWIP_MALLOC_MEMPOOL */
28
29#ifndef LWIP_PBUF_MEMPOOL
30/* This treats "pbuf pools" just like any other pool.
31 * Allocates buffers for a pbuf struct AND a payload size */
32#define LWIP_PBUF_MEMPOOL(name, num, payload, desc) LWIP_MEMPOOL(name, num, (LWIP_MEM_ALIGN_SIZE(sizeof(struct pbuf)) + LWIP_MEM_ALIGN_SIZE(payload)), desc)
33#endif /* LWIP_PBUF_MEMPOOL */
