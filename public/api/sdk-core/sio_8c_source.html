34void sio_init(u32 baudrate, u8 lcr_ueps, u8 lcr_upen, u8 lcr_usbl, u8 lcr_umode)
35{
36 u32 brd; /* Baud rate divisor. */
37 u8 bclk = 0; /* Baud rate generator clock. */
38
39 _sw(LCR_SCS_VAL | ((lcr_ueps & 1) << 4) | ((lcr_upen & 1) << 3) |
40 ((lcr_usbl & 1) << 2) | (lcr_umode & 1),
41 SIO_LCR);
42
43 /* Disable all interrupts. */
44 _sw(0, SIO_IER);
45
46 /* Reset the FIFOs. */
47 _sw(SIO_FCR_FRSTE | SIO_FCR_RFRST | SIO_FCR_TFRST, SIO_FCR);
48 /* Enable them. */
49 _sw(0, SIO_FCR);
50
51 brd = CPUCLK / (baudrate * 256);
52
53 while ((brd >= 256) && (++bclk < 4))
54 brd /= 4;
55
56 _sw((bclk << 8) | brd, SIO_BGR);
57}

59static u8 ___last_sio_putc = 0;
60int sio_putc(int c)
61{
62 if ((c == '\n') && (___last_sio_putc != '\r')) {
63 // hack: if the character to be outputted is a '\n'
64 // and the previously-outputted character is not a '\r',
65 // output a '\r' first.
66 sio_putc('\r');
67 }
68
69 /* Block until we're ready to transmit. */
70 while ((_lw(SIO_ISR) & 0xf000) == 0x8000)
71 ;
72
73 _sb(c, SIO_TXFIFO);
74 ___last_sio_putc = c;
75 return c;
76}

78int sio_getc()
79{
80 /* Do we have something in the RX FIFO? */
81 if (_lw(SIO_ISR) & 0xf00) {
82 u8 b = _lb(SIO_RXFIFO);
83 _sw(7, SIO_ISR);
84 return b;
85 }
86
87 /* Return EOF. */
88 return -1;
89}

91int sio_getc_block()
92{
93 /* Do we have something in the RX FIFO? */
94 while (!(_lw(SIO_ISR) & 0xf00))
95 ;
96 return sio_getc();
97}

110size_t sio_read(void *buf, size_t size)
111{
112 u8 *p = (u8 *)buf;
113 size_t i;
114
115 for (i = 0; i < size; i++) {
116 int c;
117
118 if ((c = sio_getc()) == -1)
119 break;
120
121 p[i] = (u8)c;
122 }
123
124 return i;
125}

138int sio_putsn(const char *str)
139{

148char *sio_gets(char *str)
149{
150 char *s = str;
151
152 while (1) {
153 int c;
154
155 c = sio_getc_block();
156 /* Check for newline. */
157 if (c == '\n' || c == '\r')
158 break;
159
160 *s++ = c;
161 }
162
163 *s = '\0';
164 return str;
165}
