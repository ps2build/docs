78/* RFC6762 section 6:
79 * To protect the network against excessive packet flooding due to software bugs
80 * or malicious attack, a Multicast DNS responder MUST NOT (except in the one
81 * special case of answering probe queries) multicast a record on a given
82 * interface until at least one second has elapsed since the last time that
83 * record was multicast on that particular interface.
84 */
85#define MDNS_MULTICAST_TIMEOUT 1000
86
87/* RFC6762 section 6:
88 * In this special case only, when responding via multicast to a probe, a
89 * Multicast DNS responder is only required to delay its transmission as
90 * necessary to ensure an interval of at least 250 ms since the last time the
91 * record was multicast on that interface.
92 */
93#define MDNS_MULTICAST_PROBE_TIMEOUT 250
94
95/* RFC6762 section 5.4:
96 * When receiving a question with the unicast-response bit set, a responder
97 * SHOULD usually respond with a unicast packet directed back to the querier.
98 * However, if the responder has not multicast that record recently (within one
99 * quarter of its TTL), then the responder SHOULD instead multicast the response
100 * so as to keep all the peer caches up to date, and to permit passive conflict
101 * detection.
102 * -> we implement a stripped down version. Depending on a timeout of 30s
103 * (25% of 120s) all QU questions are send via multicast or unicast.
104 */
105#define MDNS_MULTICAST_TIMEOUT_25TTL 30000
