33void __fdman_init()
34{
35 ee_sema_t sema;
36
37 /* Initialize mutex */
38 sema.init_count = 1;
39 sema.max_count = 1;
40 sema.option = 0;
41 __fdman_sema = CreateSema(&sema);
42 if (__fdman_sema <= 0) {
43 __fdman_sema = 0;
44 abort();
45 }
46
47 /* Initialize descriptor data*/
48 memset(__descriptor_data_pool, 0, sizeof(__descriptormap_type) *__FILENO_MAX);
49 /* Initialize descriptor map*/
50 memset(__descriptormap, 0, sizeof(__descriptormap_type*)*__FILENO_MAX);
51
52 // We assume STDIN_FILENO, STDOUT_FILENO, STDERR_FILENO are initialized
53 __descriptormap[STDIN_FILENO] = &__descriptor_data_pool[STDIN_FILENO];
54 __libcglue_init_stdio(&(__descriptormap[STDIN_FILENO]->info), 0);
55
56 __descriptormap[STDOUT_FILENO] = &__descriptor_data_pool[STDOUT_FILENO];
57 __libcglue_init_stdio(&(__descriptormap[STDOUT_FILENO]->info), 1);
58
59 __descriptormap[STDERR_FILENO] = &__descriptor_data_pool[STDERR_FILENO];
60 __libcglue_init_stdio(&(__descriptormap[STDERR_FILENO]->info), 1 /* ioman doesn't have 2 (stderr) by default, so just use 1 (stdout) */);
61}

71int __fdman_get_new_descriptor()
72{
73 int i = 0;
74
75 WaitSema(__fdman_sema); /* lock here to make thread safe */
76 for (i = 0; i < __FILENO_MAX; i++) {
77 if (__descriptormap[i] == NULL) {
78 __descriptormap[i] = &__descriptor_data_pool[i];
79 __descriptormap[i]->ref_count++;
80 SignalSema(__fdman_sema); /* release lock */
81 return i;
82 }
83 }
84 SignalSema(__fdman_sema); /* release lock */
85
86 errno = ENOMEM;
87 return -1;
88}

130void __fdman_release_descriptor(int fd)
131{
132 if (!__IS_FD_VALID(fd)) {
133 errno = EBADF;
134 return;
135 }
136
137 __descriptormap[fd]->ref_count--;
138
139 if (__descriptormap[fd]->ref_count == 0) {
140 memset(__descriptormap[fd], 0, sizeof(__descriptormap_type));
141 }
142 __descriptormap[fd] = NULL;
143}
