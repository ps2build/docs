72/* Definitions of the bits in an Internet address integer.
73
74 On subnets, host and network parts are found according to
75 the subnet mask, not these masks. */
76#define IP_CLASSA(a) ((((u32_t)(a)) & 0x80000000UL) == 0)
77#define IP_CLASSA_NET 0xff000000
78#define IP_CLASSA_NSHIFT 24
79#define IP_CLASSA_HOST (0xffffffff & ~IP_CLASSA_NET)
80#define IP_CLASSA_MAX 128
81
82#define IP_CLASSB(a) ((((u32_t)(a)) & 0xc0000000UL) == 0x80000000UL)
83#define IP_CLASSB_NET 0xffff0000
84#define IP_CLASSB_NSHIFT 16
85#define IP_CLASSB_HOST (0xffffffff & ~IP_CLASSB_NET)
86#define IP_CLASSB_MAX 65536
87
88#define IP_CLASSC(a) ((((u32_t)(a)) & 0xe0000000UL) == 0xc0000000UL)
89#define IP_CLASSC_NET 0xffffff00
90#define IP_CLASSC_NSHIFT 8
91#define IP_CLASSC_HOST (0xffffffff & ~IP_CLASSC_NET)
92
93#define IP_CLASSD(a) (((u32_t)(a) & 0xf0000000UL) == 0xe0000000UL)
94#define IP_CLASSD_NET 0xf0000000 /* These ones aren't really */
95#define IP_CLASSD_NSHIFT 28 /* net and host fields, but */
96#define IP_CLASSD_HOST 0x0fffffff /* routing needn't know. */
97#define IP_MULTICAST(a) IP_CLASSD(a)
98
99#define IP_EXPERIMENTAL(a) (((u32_t)(a) & 0xf0000000UL) == 0xf0000000UL)
100#define IP_BADCLASS(a) (((u32_t)(a) & 0xf0000000UL) == 0xf0000000UL)
