27void _libcglue_timezone_update_impl()
28{
29 /* Initialize timezone from PS2 OSD configuration */
30 int tzOffset = configGetTimezone();
31 int tzOffsetAbs = tzOffset < 0 ? -tzOffset : tzOffset;
32 int hours = tzOffsetAbs / 60;
33 int minutes = tzOffsetAbs - hours * 60;
34 int daylight = configIsDaylightSavingEnabled();
35 static char tz[28];
36 #pragma GCC diagnostic push
37 #pragma GCC diagnostic ignored "-Wformat-overflow"
38 snprintf(tz, sizeof(tz), "GMT%s%02i:%02i%s", tzOffset < 0 ? "+" : "-", hours, minutes, daylight ? "DST" : "");
39 #pragma GCC diagnostic pop
40 setenv("TZ", tz, 1);
41}
