92static void
93tcpip_mbox_fetch(sys_mbox_t *mbox, void **msg)
94{
95 u32_t sleeptime, res;
96
97again:
98 LWIP_ASSERT_CORE_LOCKED();
99
100 sleeptime = sys_timeouts_sleeptime();
101 if (sleeptime == SYS_TIMEOUTS_SLEEPTIME_INFINITE) {
102 UNLOCK_TCPIP_CORE();
103 sys_arch_mbox_fetch(mbox, msg, 0);
104 LOCK_TCPIP_CORE();
105 return;
106 } else if (sleeptime == 0) {
107 sys_check_timeouts();
108 /* We try again to fetch a message from the mbox. */
109 goto again;
110 }
111
112 UNLOCK_TCPIP_CORE();
113 res = sys_arch_mbox_fetch(mbox, msg, sleeptime);
114 LOCK_TCPIP_CORE();
115 if (res == SYS_ARCH_TIMEOUT) {
116 /* If a SYS_ARCH_TIMEOUT value is returned, a timeout occurred
117 before a message could be fetched. */
118 sys_check_timeouts();
119 /* We try again to fetch a message from the mbox. */
120 goto again;
121 }
122}

135static void
136tcpip_thread(void *arg)
137{
138 struct tcpip_msg *msg;
139 LWIP_UNUSED_ARG(arg);
140
141 LWIP_MARK_TCPIP_THREAD();
142
143 LOCK_TCPIP_CORE();
144 if (tcpip_init_done != NULL) {
145 tcpip_init_done(tcpip_init_done_arg);
146 }
147
148 while (1) { /* MAIN Loop */
149 LWIP_TCPIP_THREAD_ALIVE();
150 /* wait for a message, timeouts are processed while waiting */
151 tcpip_mbox_fetch(&tcpip_mbox, (void **)&msg);
152 if (msg == NULL) {
153 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: invalid message: NULL\n"));
154 LWIP_ASSERT("tcpip_thread: invalid message", 0);
155 continue;
156 }
157 tcpip_thread_handle_msg(msg);
158 }
159}

161/* Handle a single tcpip_msg
162 * This is in its own function for access by tests only.
163 */
164static void
165tcpip_thread_handle_msg(struct tcpip_msg *msg)
166{
167 switch (msg->type) {
168#if !LWIP_TCPIP_CORE_LOCKING
169 case TCPIP_MSG_API:
170 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: API message %p\n", (void *)msg));
171 msg->msg.api_msg.function(msg->msg.api_msg.msg);
172 break;
173 case TCPIP_MSG_API_CALL:
174 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: API CALL message %p\n", (void *)msg));
175 msg->msg.api_call.arg->err = msg->msg.api_call.function(msg->msg.api_call.arg);
176 sys_sem_signal(msg->msg.api_call.sem);
177 break;
178 case TCPIP_MSG_CALLBACK_STATIC_WAIT:
179 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: CALLBACK WAIT message %p\n", (void *)msg));
180 msg->msg.cb_wait.function(msg->msg.cb_wait.ctx);
181 sys_sem_signal(msg->msg.cb_wait.sem);
182 break;
183#endif /* !LWIP_TCPIP_CORE_LOCKING */
184
185#if !LWIP_TCPIP_CORE_LOCKING_INPUT
186 case TCPIP_MSG_INPKT:
187 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: PACKET %p\n", (void *)msg));
188 if (msg->msg.inp.input_fn(msg->msg.inp.p, msg->msg.inp.netif) != ERR_OK) {
189 pbuf_free(msg->msg.inp.p);
190 }
191 memp_free(MEMP_TCPIP_MSG_INPKT, msg);
192 break;
193#endif /* !LWIP_TCPIP_CORE_LOCKING_INPUT */
194
195#if LWIP_TCPIP_TIMEOUT && LWIP_TIMERS
196 case TCPIP_MSG_TIMEOUT:
197 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: TIMEOUT %p\n", (void *)msg));
198 sys_timeout(msg->msg.tmo.msecs, msg->msg.tmo.h, msg->msg.tmo.arg);
199 memp_free(MEMP_TCPIP_MSG_API, msg);
200 break;
201 case TCPIP_MSG_UNTIMEOUT:
202 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: UNTIMEOUT %p\n", (void *)msg));
203 sys_untimeout(msg->msg.tmo.h, msg->msg.tmo.arg);
204 memp_free(MEMP_TCPIP_MSG_API, msg);
205 break;
206#endif /* LWIP_TCPIP_TIMEOUT && LWIP_TIMERS */
207
208 case TCPIP_MSG_CALLBACK:
209 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: CALLBACK %p\n", (void *)msg));
210 msg->msg.cb.function(msg->msg.cb.ctx);
211 memp_free(MEMP_TCPIP_MSG_API, msg);
212 break;
213
214 case TCPIP_MSG_CALLBACK_STATIC:
215 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: CALLBACK_STATIC %p\n", (void *)msg));
216 msg->msg.cb.function(msg->msg.cb.ctx);
217 break;
218
219 default:
220 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_thread: invalid message: %d\n", msg->type));
221 LWIP_ASSERT("tcpip_thread: invalid message", 0);
222 break;
223 }
224}

253err_t
254tcpip_inpkt(struct pbuf *p, struct netif *inp, netif_input_fn input_fn)
255{
256#if LWIP_TCPIP_CORE_LOCKING_INPUT
257 err_t ret;
258 LWIP_DEBUGF(TCPIP_DEBUG, ("tcpip_inpkt: PACKET %p/%p\n", (void *)p, (void *)inp));
259 LOCK_TCPIP_CORE();
260 ret = input_fn(p, inp);
261 UNLOCK_TCPIP_CORE();
262 return ret;
263#else /* LWIP_TCPIP_CORE_LOCKING_INPUT */
264 struct tcpip_msg *msg;
265
266 LWIP_ASSERT("Invalid mbox", sys_mbox_valid_val(tcpip_mbox));
267
268 msg = (struct tcpip_msg *)memp_malloc(MEMP_TCPIP_MSG_INPKT);
269 if (msg == NULL) {
270 return ERR_MEM;
271 }
272
273 msg->type = TCPIP_MSG_INPKT;
274 msg->msg.inp.p = p;
275 msg->msg.inp.netif = inp;
276 msg->msg.inp.input_fn = input_fn;
277 if (sys_mbox_trypost(&tcpip_mbox, msg) != ERR_OK) {
278 memp_free(MEMP_TCPIP_MSG_INPKT, msg);
279 return ERR_MEM;
280 }
281 return ERR_OK;
282#endif /* LWIP_TCPIP_CORE_LOCKING_INPUT */
283}

486err_t
487tcpip_api_call(tcpip_api_call_fn fn, struct tcpip_api_call_data *call)
488{
489#if LWIP_TCPIP_CORE_LOCKING
490 err_t err;
491 LOCK_TCPIP_CORE();
492 err = fn(call);
493 UNLOCK_TCPIP_CORE();
494 return err;
495#else /* LWIP_TCPIP_CORE_LOCKING */
496 TCPIP_MSG_VAR_DECLARE(msg);
497
498#if !LWIP_NETCONN_SEM_PER_THREAD
499 err_t err = sys_sem_new(&call->sem, 0);
500 if (err != ERR_OK) {
501 return err;
502 }
503#endif /* LWIP_NETCONN_SEM_PER_THREAD */
504
505 LWIP_ASSERT("Invalid mbox", sys_mbox_valid_val(tcpip_mbox));
506
507 TCPIP_MSG_VAR_ALLOC(msg);
508 TCPIP_MSG_VAR_REF(msg).type = TCPIP_MSG_API_CALL;
509 TCPIP_MSG_VAR_REF(msg).msg.api_call.arg = call;
510 TCPIP_MSG_VAR_REF(msg).msg.api_call.function = fn;
511#if LWIP_NETCONN_SEM_PER_THREAD
512 TCPIP_MSG_VAR_REF(msg).msg.api_call.sem = LWIP_NETCONN_THREAD_SEM_GET();
513#else /* LWIP_NETCONN_SEM_PER_THREAD */
514 TCPIP_MSG_VAR_REF(msg).msg.api_call.sem = &call->sem;
515#endif /* LWIP_NETCONN_SEM_PER_THREAD */
516 sys_mbox_post(&tcpip_mbox, &TCPIP_MSG_VAR_REF(msg));
517 sys_arch_sem_wait(TCPIP_MSG_VAR_REF(msg).msg.api_call.sem, 0);
518 TCPIP_MSG_VAR_FREE(msg);
519
520#if !LWIP_NETCONN_SEM_PER_THREAD
521 sys_sem_free(&call->sem);
522#endif /* LWIP_NETCONN_SEM_PER_THREAD */
523
524 return call->err;
525#endif /* LWIP_TCPIP_CORE_LOCKING */
526}

618err_t
619tcpip_callback_wait(tcpip_callback_fn function, void *ctx)
620{
621#if LWIP_TCPIP_CORE_LOCKING
622 LOCK_TCPIP_CORE();
623 function(ctx);
624 UNLOCK_TCPIP_CORE();
625 return ERR_OK;
626#else /* LWIP_TCPIP_CORE_LOCKING */
627 err_t err;
628 sys_sem_t sem;
629 struct tcpip_msg msg;
630
631 LWIP_ASSERT("Invalid mbox", sys_mbox_valid_val(tcpip_mbox));
632
633 err = sys_sem_new(&sem, 0);
634 if (err != ERR_OK) {
635 return err;
636 }
637
638 msg.type = TCPIP_MSG_CALLBACK_STATIC_WAIT;
639 msg.msg.cb_wait.function = function;
640 msg.msg.cb_wait.ctx = ctx;
641 msg.msg.cb_wait.sem = &sem;
642 sys_mbox_post(&tcpip_mbox, &msg);
643 sys_arch_sem_wait(&sem, 0);
644 sys_sem_free(&sem);
645 return ERR_OK;
646#endif /* LWIP_TCPIP_CORE_LOCKING */
647}

658void
659tcpip_init(tcpip_init_done_fn initfunc, void *arg)
660{
661 lwip_init();
662
663 tcpip_init_done = initfunc;
664 tcpip_init_done_arg = arg;
665 if (sys_mbox_new(&tcpip_mbox, TCPIP_MBOX_SIZE) != ERR_OK) {
666 LWIP_ASSERT("failed to create tcpip_thread mbox", 0);
667 }
668#if LWIP_TCPIP_CORE_LOCKING
669 if (sys_mutex_new(&lock_tcpip_core) != ERR_OK) {
670 LWIP_ASSERT("failed to create lock_tcpip_core", 0);
671 }
672#endif /* LWIP_TCPIP_CORE_LOCKING */
673
674 sys_thread_new(TCPIP_THREAD_NAME, tcpip_thread, NULL, TCPIP_THREAD_STACKSIZE, TCPIP_THREAD_PRIO);
675}
