27int SifInitIopHeap()
28{
29 int res;
30 if (HasIopRebootedSinceLastCall())
31 SifExitIopHeap();
32
33 if (_ih_cd.server)
34 return 0;
35
36 sceSifInitRpc(0);
37
38 while ((res = sceSifBindRpc(&_ih_cd, 0x80000003, 0)) >= 0 && !_ih_cd.server)
39 nopdelay();
40
41 if (res < 0)
42 return -E_SIF_RPC_BIND;
43
44 return 0;
45}

52void *SifAllocIopHeap(int size)
53{

90int SifLoadIopHeap(const char *path, void *addr)

92 struct _iop_load_heap_arg arg;

94 if (SifInitIopHeap() < 0)

97 arg.p.addr = addr;

100 if (sceSifCallRpc(&_ih_cd, 3, 0, &arg, sizeof arg, &arg, 4, NULL, NULL) < 0)
