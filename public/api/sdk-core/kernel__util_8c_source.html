26s32 WaitSemaEx(s32 semaid, int signal, u64 *timeout)
27{
28 int ret;
29 int timerid;

31 timerid = -1;
32
33 // TODO: other values NYI
34 if (signal != 1)
35 {
36 return -100;
37 }
38
39 if (timeout != NULL && *timeout == 0)
40 {
41 ret = PollSema(semaid);
42 if (ret < 0)
43 {
44 return ret;
45 }
46 return semaid;
47 }
48
49 if (timeout != NULL)
50 {
51 timerid = SetTimerAlarm(USec2TimerBusClock(*timeout), &WaitSemaEx_callback, (void *)GetThreadId());
52 }
53
54 ret = WaitSema(semaid);
55
56 if (timerid >= 0)
57 {
58 ReleaseTimerAlarm(timerid);
59 }
60
61 return ret;
