35void SetupRomInfo(void)
36{
37 void *iop_addr;
38 SifRpcReceiveData_t rdata;
39 u8 buf[64] __attribute__((__aligned__(64)));
40
41 /* ROMVER needs to be read from the IOP due to PPCIOP region patching. */
42 /* only read in the romname the first time */
43 if (g_rom0_info_data.m_romver[0])
44 return;
45
46 /* SYSMEM allocates in units of 256. */
47 if (!(iop_addr = SifAllocIopHeap(256)))
48 return;
49
50 /* ROMVER is known to be 16 bytes. */
51 if (SifLoadIopHeap("rom:ROMVER", iop_addr) < 0)
52 {
53 SifFreeIopHeap(iop_addr);
54 return;
55 }
56 SyncDCache(buf, buf + sizeof(buf));
57 if (sceSifGetOtherData(&rdata, iop_addr, buf, sizeof(buf), 0) < 0)
58 {
59 SifFreeIopHeap(iop_addr);
60 return;
61 }
62 memcpy(g_rom0_info_data.m_romver, UNCACHED_SEG(buf), sizeof(g_rom0_info_data.m_romver));
63
64 SifFreeIopHeap(iop_addr);
65}

67char *GetRomName(char *romname)
68{
69 SetupRomInfo();
70 /* Explicitly copy 14 bytes to the buffer, without NULL terminator */
71 memcpy(romname, g_rom0_info_data.m_romver, sizeof(g_rom0_info_data.m_romver));
72 return romname;
73}
