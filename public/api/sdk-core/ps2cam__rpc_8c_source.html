217int PS2CamReadPacket(int handle)
218{
219 int *ret;
220 SifRpcReceiveData_t rdata;
221
222 WaitSema(sem);
223
224 ret = (int *)&data[0];
225
226 ret[0] = handle;
227
228 sceSifCallRpc(&cdata, PS2CAM_RPC_READPACKET, 0, (void*)(&data[0]),4,(void*)(&data[0]),4*2,0,0);
229
230
231 if(ret[0] < 0) return ret[0];
232
233 SyncDCache(campacket, (u8 *)(&campacket[0]) + sizeof(campacket));
234 if (sceSifGetOtherData(&rdata, (void *)ret[1], &campacket[0], sizeof(campacket), 0) < 0)
235 return -1;
236
237 //if i add a printf here, the ps2 will exit to sony's explorer
238 SignalSema(sem);
239 return ret[0];
240}

278int PS2CamExtractFrame(int handle, char *buffer, int bufsize)
279{
280 static int capturing;
281 static int pos;
282 static int pic_size;
283
284 (void)bufsize;
285
286 pos = 0;
287 capturing = 0;
288
289
290
291 while(1)
292 {
293 static EYETOY_FRAME_HEAD *head;
294 static int ret;
295
296 ret = PS2CamReadPacket(handle);
297
298 // printf("packet =%d\n",ret);
299 //if read has a error return it
300 if(ret < 0) return ret;
301
302 head = (EYETOY_FRAME_HEAD *)UNCACHED_SEG(&campacket[0]);
303
304 if(head->magic2==0xff && head->magic3==0xff && ret!=0)
305 {
306 if(head->type==0x50 && head->frame==0x00)
307 {
308 // start of frame
309 // printf("SOF(%d)\n",ret);
310 if(capturing == 1)
311 {
312 return 0;
313 }
314
315 memcpy(&buffer[pos], UNCACHED_SEG(&campacket[16]), ret-16 );
316 pos += (ret-16);
317 capturing = 1;
318
319 }
320 else if(head->type==0x50 && head->frame==0x01)
321 {
322 // frame head with no data
323 // so do nothing
324
325 }
326 else if(head->type==0x51 && capturing==1)
327 {
328 //end of frame
329 // printf("EOF(%d)\n",ret);
330 pic_size = (int)(((head->Lo) + ((int)(head->Hi)<<8))<<3);
331
332
333 if(pos != pic_size)
334 {
335 return 0;
336 }
337 else
338 {
339 return pic_size;
340 }
341 }
342 else
343 {
344 //if it doesnt fit in those pos then it must be data
345 if(capturing==1)
346 {
347 memcpy(&buffer[pos], UNCACHED_SEG(&campacket[0]), ret );
348 pos += (ret);
349 }
350 }
351 }
352 else
353 {
354 // it must be data
355 if(capturing==1 && ret !=0)
356 {
357 memcpy(&buffer[pos], UNCACHED_SEG(&campacket[0]), ret );
358 pos += (ret);
359 }
360 }
361 }
362
363}
