28// Real, confirmed bug (2026-09-07): __attribute__((constructor)) never ran
29// anywhere in this SDK, on real hardware or any emulator - "if (_init)
30// _init();" below was always a no-op. _init/__libc_init_array are
31// normally provided by gcc's own crtbegin.o/crtend.o, which -nostartfiles
32// (required here - this file *is* the startup code) never links. This
33// compiler emits the legacy .ctors section (confirmed via objdump - not
34// .init_array), so crtbegin.o's own __do_global_ctors_aux is what would
35// have walked it; without that object, nothing does. __CTOR_LIST__/
36// __CTOR_END__ (provided by the linkfile, see its own comment) bound the
37// section directly, sidestepping the need for crtbegin.o's own -1
38// start-sentinel/0 end-terminator convention entirely. Real GCC .ctors
39// calling order is reverse-array (highest address/lowest priority
40// first), matching __do_global_ctors_aux's own real behavior.
41extern void (*__CTOR_LIST__[])(void);
42extern void (*__CTOR_END__[])(void);
43
44// clang emits .init_array instead of .ctors for global/static C++
45// constructors - see the linkfile's own comment on __init_array_start/
46// __init_array_end. Unlike .ctors (walked in reverse), .init_array runs
47// in forward array order per the Itanium C++ ABI.
48extern void (*__init_array_start[])(void);
49extern void (*__init_array_end[])(void);
50
51// Real, confirmed bug: any global/static C++ object with a non-trivial
52// destructor fails to link with "undefined reference to `__dso_handle`".
53// GCC emits a __cxa_atexit(dtor, obj, &__dso_handle) call at that
54// object's construction point; __dso_handle is normally defined by
55// gcc's own crtbegin.o, which -nostartfiles (required here - this file
56// *is* the startup code, see above) never links. A statically-linked
57// PS2 ELF is always exactly one "module" (no dlclose, no shared
58// objects), so the value only needs to be a stable, unique address -
59// same minimal fix used by other no-crtbegin bare-metal targets
60// (avr-libc and similar). weak so a real crtbegin.o's own definition
61// would still win if some future path ever links one instead.
62void *__dso_handle __attribute__((weak)) = &__dso_handle;

64static void _call_global_ctors(void)
65{
66 int count = __CTOR_END__ - __CTOR_LIST__;
67 for (int i = count - 1; i >= 0; i--)
68 __CTOR_LIST__[i]();
69
70 int init_array_count = __init_array_end - __init_array_start;
71 for (int i = 0; i < init_array_count; i++)
72 __init_array_start[i]();
73}

85/*
86 * First function to be called by the loader
87 * This function sets up the stack and heap.
88 * DO NOT USE THE STACK IN THIS FUNCTION!
89 */
90void __start(struct sargs_start *pargs)
91{
92 __asm__ __volatile__(
93 "# Clear bss area \n"
94 ".set noat \n"
95 "la $2, _fbss \n"
96 "la $3, _end \n"
97 "1: \n"
98 "sltu $1, $2, $3 \n"
99 "beq $1, $0, 2f \n"
100 "nop \n"
101 "sq $0, ($2) \n"
102 "addiu $2, $2, 16 \n"
103 "j 1b \n"
104 "nop \n"
105 "2: \n"
106 " \n"
107 "# Save first argument \n"
108 "sw %1, %0 \n"
109 " \n"
110 "# SetupThread \n"
111 "la $4, _gp \n"
112 "la $5, _stack \n"
113 "la $6, _stack_size \n"
114 "la $7, args \n"
115 "la $8, ExitThread \n"
116 "move $gp, $4 \n"
117 "addiu $3, $0, 60 \n"
118 "syscall \n"
119 "move $sp, $2 \n"
120 " \n"
121 "# Jump to _main \n"
122 "j %2 \n"
123 ".set at \n"
124 : /* No outputs. */
125 : "m"(args_start), "r"(pargs), "Csy"(_main)
126 : "1", "2", "3", "4", "5", "6", "7", "8");
127}

129/*
130 * Intermediate function between _start and main, stack can be used as normal.
131 */
132static void _main()
133{
134 int retval;
135 struct sargs *pa;
136
137 // initialize heap
138 SetupHeap(&_end, (int)&_heap_size);
139
140 // writeback data cache
141 FlushCache(0);
142
143 // Capability to override 32MiB on DESR/DVR models
144 // NOTE: this call can restart the application
145 if (_ps2sdk_memory_init)
146 _ps2sdk_memory_init();
147
148 // Initialize the kernel (Apply necessary patches).
149 _InitSys();
150
151 // Use arguments sent through start if sent (by ps2link for instance)
152 pa = &args;
153 if (args.argc == 0 && args_start != NULL && args_start->args.argc != 0)
154 pa = &args_start->args;
155
156 // call libcglue argument parsing
157 _libcglue_args_parse(pa->argc, pa->argv);
158
159 // initialize libcglue
160 _libcglue_init();
161
162 // Enable interruts
163 EI();
164
165 // call global constructors - see _call_global_ctors's own comment
166 // for why this replaces the old (always-dead) "if (_init) _init();"
167 _call_global_ctors();
168
169 // call main
170 retval = main(pa->argc, pa->argv);
171
172 // call global destructors (weak)
173 if (_fini)
174 _fini();
175
176 // uninitialize libcglue
177 _libcglue_deinit();
178
179 // Exit
180 Exit(retval);
181}
