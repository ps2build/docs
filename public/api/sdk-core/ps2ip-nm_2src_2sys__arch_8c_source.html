39static arch_message *try_alloc_msg(void)
40{
41 arch_message *message;
42 int OldState;
43
44 if(PollSema(MsgCountSema)==0)
45 {
46 CpuSuspendIntr(&OldState);
47
48 message = free_head;
49 free_head = free_head->next;
50
51 CpuResumeIntr(OldState);
52 }else message=NULL;
53
54 return message;
55}

74static void free_msg(arch_message *msg)
75{
76 int oldIntr;
77
78 CpuSuspendIntr(&oldIntr);
79
80 msg->next = free_head;
81 free_head = msg;
82
83 CpuResumeIntr(oldIntr);
84 SignalSema(MsgCountSema);
85}

94static u32_t
95ComputeTimeDiff(iop_sys_clock_t* pStart,iop_sys_clock_t* pEnd)
96{
97 iop_sys_clock_t Diff;
98 u32 iSec, iUSec, iDiff;
99
100 Diff.lo=pEnd->lo-pStart->lo;
101 Diff.hi=pEnd->hi-pStart->hi - (pStart->lo>pEnd->lo);
102
103 SysClock2USec(&Diff, &iSec, &iUSec);
104 iDiff=(iSec*1000)+(iUSec/1000);
105
106 return((iDiff!=0)?iDiff:1);
107}

110sys_thread_t sys_thread_new(const char *name, lwip_thread_fn thread, void *arg, int stacksize, int prio)
111{
112 iop_thread_t thp;
113 int tid, rv;
114
115 (void)name;
116
117 thp.attr = TH_C;
118 thp.option = (u32)"PS2IP";
119 thp.thread = thread;
120 thp.stacksize = stacksize;
121 thp.priority = prio;
122
123 if((tid = CreateThread(&thp)) < 0)
124 {
125 dbgprintf("sys_thread_new: CreateThread failed, EC: %d\n", tid);
126 return ERR_MEM;
127 }
128
129 if((rv = StartThread(tid, arg)) < 0)
130 {
131 dbgprintf("sys_thread_new: StartThread failed, EC: %d\n", rv);
132 DeleteThread(tid);
133 return ERR_MEM;
134 }
135
136 dbgprintf("sys_thread_new: thread %d\n", tid);
137
138 return((sys_thread_t) tid);
139}

205static u32_t sys_arch_mbox_fetch_internal(sys_mbox_t pMBox, void** ppvMSG, u32_t TimeElaspedout, char block)
206{
207 void *pmsg;
208 u32_t TimeElasped = 0;
209 iop_sys_clock_t ClockTicks;
210 iop_sys_clock_t Start;
211 iop_sys_clock_t End;
212 int result;
213
214 pmsg = 0;
215 result = -1;
216 if(block){
217 int iPID;
218
219 iPID=GetThreadId();
220
221 if(TimeElaspedout > 0)
222 {
223 GetSystemTime(&Start);
224 USec2SysClock(TimeElaspedout * 1000, &ClockTicks);
225 SetAlarm(&ClockTicks, &TimeoutHandler, (void*)iPID);
226 }
227
228 if((result=ReceiveMbx(&pmsg, pMBox))!= 0) { return(SYS_ARCH_TIMEOUT); }
229
230 if(TimeElaspedout > 0)
231 {
232 CancelAlarm(TimeoutHandler,(void*)iPID);
233 GetSystemTime(&End);
234
235 TimeElasped = ComputeTimeDiff(&Start, &End);
236 }
237 }
238 else{
239 TimeElasped=((result=PollMbx(&pmsg, pMBox))!=0)?SYS_MBOX_EMPTY:0;
240 }
241
242 if(result==0){
243 if (pmsg != NULL)
244 {
245 *ppvMSG = ((arch_message *)pmsg)->sys_msg;
246 free_msg((arch_message *) pmsg);
247 }
248 }
249
250 //Return the number of msec waited.
251 return TimeElasped;
252}

289u32_t sys_arch_sem_wait(sys_sem_t *sem, u32_t timeout)
290{
291 u32_t result;
292
293 //Wait TimeElaspedout msec for the Sema to receive a signal.
294 if(timeout==0)
295 {
296 //Wait with no timeouts.
297 result=(WaitSema(*sem)==0)?0:SYS_ARCH_TIMEOUT;
298 }
299 else if(timeout==1)
300 {
301 //Poll.
302 result=(PollSema(*sem)==0)?0:SYS_ARCH_TIMEOUT;
303 }
304 else
305 {
306 //Use alarm to timeout.
307 iop_sys_clock_t ClockTicks;
308 iop_sys_clock_t Start;
309 iop_sys_clock_t End;
310 int iPID=GetThreadId();
311 u32_t u32WaitTime;
312
313 GetSystemTime(&Start);
314 USec2SysClock(timeout*1000,&ClockTicks);
315 SetAlarm(&ClockTicks, &TimeoutHandler, (void*)iPID);
316
317 if(WaitSema(*sem)==0){
318 CancelAlarm(TimeoutHandler,(void*)iPID);
319 GetSystemTime(&End);
320
321 u32WaitTime=ComputeTimeDiff(&Start, &End);
322 result=(u32WaitTime<=timeout)?u32WaitTime:timeout;
323 }else result=SYS_ARCH_TIMEOUT;
324 }
325
326 return result;
327}

339void sys_init(void)
340{
341 arch_message *prev;
342 unsigned int i;
343 iop_sema_t sema;
344
345 sema.initial=sema.max=SYS_MAX_MESSAGES;
346 sema.attr=sema.option=0;
347 MsgCountSema=CreateSema(&sema);
348
349 free_head = &msg_pool[0];
350 prev = &msg_pool[0];
351
352 for(i = 1; i < SYS_MAX_MESSAGES; i++)
353 {
354 prev->next = &msg_pool[i];
355 prev = &msg_pool[i];
356 }
357
358 //NULL-terminate free message list
359 prev->next = NULL;
360}
