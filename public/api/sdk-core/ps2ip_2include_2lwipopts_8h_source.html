59/* MEM_ALIGNMENT: must equal the alignment libc's malloc returns,
60 because pbuf_alloc(PBUF_RAM) computes
61 payload = LWIP_MEM_ALIGN(p + SIZEOF_STRUCT_PBUF + offset)
62 inside an allocation sized assuming p is already MEM_ALIGNMENT-
63 aligned. If MEM_ALIGNMENT > newlib's alignment the payload pointer
64 overruns the chunk and corrupts the next-chunk header on the
65 freelist (TLB misses inside _malloc_r / _free_r).
66
67 16 is the contract baked into the toolchain configuration. See
68 newlib/newlib/configure.host:
69
70 mips64r5900*)
71 machine_dir=r5900
72 newlib_cflags="${newlib_cflags} -DMALLOC_ALIGNMENT=16"
73 ;;
74
75 so for the mips64r5900el-ps2-elf target newlib is built with
76 -DMALLOC_ALIGNMENT=16, which sets _mallocr.c's MALLOC_ALIGNMENT
77 directly (overriding the SIZE_SZ-derived default of 8). 16 is also
78 what samples/malloc_stress observes empirically.
79
80 The historical value here was 64 with a comment about "the EE cache
81 design" (SP193). That was over-cautious: PBUF_POOL pbufs (the only
82 ones touched by IOP->EE DMA + cache invalidate) come from memp's
83 static pools and are always 64-byte aligned regardless of
84 MEM_ALIGNMENT; PBUF_RAM pbufs (TX, ARP, DNS, ...) only see DMA
85 writeback, where misalignment harmlessly over-flushes extra cache
86 lines. The IOP-side lwipopts has used MEM_ALIGNMENT=4 forever for
87 the same reason. */
88#define MEM_ALIGNMENT 16
