79void
80icmp_input(struct pbuf *p, struct netif *inp)
81{
82 u8_t type;
83#ifdef LWIP_DEBUG
84 u8_t code;
85#endif /* LWIP_DEBUG */
86 struct icmp_echo_hdr *iecho;
87 const struct ip_hdr *iphdr_in;
88 u16_t hlen;
89 const ip4_addr_t *src;
90
91 ICMP_STATS_INC(icmp.recv);
92 MIB2_STATS_INC(mib2.icmpinmsgs);
93
94 iphdr_in = ip4_current_header();
95 hlen = IPH_HL_BYTES(iphdr_in);
96 if (hlen < IP_HLEN) {
97 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: short IP header (%"S16_F" bytes) received\n", hlen));
98 goto lenerr;
99 }
100 if (p->len < sizeof(u16_t) * 2) {
101 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: short ICMP (%"U16_F" bytes) received\n", p->tot_len));
102 goto lenerr;
103 }
104
105 type = *((u8_t *)p->payload);
106#ifdef LWIP_DEBUG
107 code = *(((u8_t *)p->payload) + 1);
108 /* if debug is enabled but debug statement below is somehow disabled: */
109 LWIP_UNUSED_ARG(code);
110#endif /* LWIP_DEBUG */
111 switch (type) {
112 case ICMP_ER:
113 /* This is OK, echo reply might have been parsed by a raw PCB
114 (as obviously, an echo request has been sent, too). */
115 MIB2_STATS_INC(mib2.icmpinechoreps);
116 break;
117 case ICMP_ECHO:
118 MIB2_STATS_INC(mib2.icmpinechos);
119 src = ip4_current_dest_addr();
120 /* multicast destination address? */
121 if (ip4_addr_ismulticast(ip4_current_dest_addr())) {
122#if LWIP_MULTICAST_PING
123 /* For multicast, use address of receiving interface as source address */
124 src = netif_ip4_addr(inp);
125#else /* LWIP_MULTICAST_PING */
126 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: Not echoing to multicast pings\n"));
127 goto icmperr;
128#endif /* LWIP_MULTICAST_PING */
129 }
130 /* broadcast destination address? */
131 if (ip4_addr_isbroadcast(ip4_current_dest_addr(), ip_current_netif())) {
132#if LWIP_BROADCAST_PING
133 /* For broadcast, use address of receiving interface as source address */
134 src = netif_ip4_addr(inp);
135#else /* LWIP_BROADCAST_PING */
136 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: Not echoing to broadcast pings\n"));
137 goto icmperr;
138#endif /* LWIP_BROADCAST_PING */
139 }
140 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: ping\n"));
141 if (p->tot_len < sizeof(struct icmp_echo_hdr)) {
142 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: bad ICMP echo received\n"));
143 goto lenerr;
144 }
145#if CHECKSUM_CHECK_ICMP
146 IF__NETIF_CHECKSUM_ENABLED(inp, NETIF_CHECKSUM_CHECK_ICMP) {
147 if (inet_chksum_pbuf(p) != 0) {
148 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: checksum failed for received ICMP echo\n"));
149 pbuf_free(p);
150 ICMP_STATS_INC(icmp.chkerr);
151 MIB2_STATS_INC(mib2.icmpinerrors);
152 return;
153 }
154 }
155#endif
156#if LWIP_ICMP_ECHO_CHECK_INPUT_PBUF_LEN
157 if (pbuf_add_header(p, hlen + PBUF_LINK_HLEN + PBUF_LINK_ENCAPSULATION_HLEN)) {
158 /* p is not big enough to contain link headers
159 * allocate a new one and copy p into it
160 */
161 struct pbuf *r;
162 u16_t alloc_len = (u16_t)(p->tot_len + hlen);
163 if (alloc_len < p->tot_len) {
164 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: allocating new pbuf failed (tot_len overflow)\n"));
165 goto icmperr;
166 }
167 /* allocate new packet buffer with space for link headers */
168 r = pbuf_alloc(PBUF_LINK, alloc_len, PBUF_RAM);
169 if (r == NULL) {
170 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: allocating new pbuf failed\n"));
171 goto icmperr;
172 }
173 if (r->len < hlen + sizeof(struct icmp_echo_hdr)) {
174 LWIP_DEBUGF(ICMP_DEBUG | LWIP_DBG_LEVEL_SERIOUS, ("first pbuf cannot hold the ICMP header\n"));
175 pbuf_free(r);
176 goto icmperr;
177 }
178 /* copy the ip header */
179 MEMCPY(r->payload, iphdr_in, hlen);
180 /* switch r->payload back to icmp header (cannot fail) */
181 if (pbuf_remove_header(r, hlen)) {
182 LWIP_ASSERT("icmp_input: moving r->payload to icmp header failed", 0);
183 pbuf_free(r);
184 goto icmperr;
185 }
186 /* copy the rest of the packet without ip header */
187 if (pbuf_copy(r, p) != ERR_OK) {
188 LWIP_DEBUGF(ICMP_DEBUG | LWIP_DBG_LEVEL_SERIOUS, ("icmp_input: copying to new pbuf failed\n"));
189 pbuf_free(r);
190 goto icmperr;
191 }
192 /* free the original p */
193 pbuf_free(p);
194 /* we now have an identical copy of p that has room for link headers */
195 p = r;
196 } else {
197 /* restore p->payload to point to icmp header (cannot fail) */
198 if (pbuf_remove_header(p, hlen + PBUF_LINK_HLEN + PBUF_LINK_ENCAPSULATION_HLEN)) {
199 LWIP_ASSERT("icmp_input: restoring original p->payload failed", 0);
200 goto icmperr;
201 }
202 }
203#endif /* LWIP_ICMP_ECHO_CHECK_INPUT_PBUF_LEN */
204 /* At this point, all checks are OK. */
205 /* We generate an answer by switching the dest and src ip addresses,
206 * setting the icmp type to ECHO_RESPONSE and updating the checksum. */
207 iecho = (struct icmp_echo_hdr *)p->payload;
208 if (pbuf_add_header(p, hlen)) {
209 LWIP_DEBUGF(ICMP_DEBUG | LWIP_DBG_LEVEL_SERIOUS, ("Can't move over header in packet\n"));
210 } else {
211 err_t ret;
212 struct ip_hdr *iphdr = (struct ip_hdr *)p->payload;
213 ip4_addr_copy(iphdr->src, *src);
214 ip4_addr_copy(iphdr->dest, *ip4_current_src_addr());
215 ICMPH_TYPE_SET(iecho, ICMP_ER);
216 p->if_idx = NETIF_NO_INDEX; /* we're reusing this pbuf, so reset its if_idx */
217#if CHECKSUM_GEN_ICMP
218 IF__NETIF_CHECKSUM_ENABLED(inp, NETIF_CHECKSUM_GEN_ICMP) {
219 /* adjust the checksum */
220 if (iecho->chksum > PP_HTONS(0xffffU - (ICMP_ECHO << 8))) {
221 iecho->chksum = (u16_t)(iecho->chksum + PP_HTONS((u16_t)(ICMP_ECHO << 8)) + 1);
222 } else {
223 iecho->chksum = (u16_t)(iecho->chksum + PP_HTONS(ICMP_ECHO << 8));
224 }
225 }
226#if LWIP_CHECKSUM_CTRL_PER_NETIF
227 else {
228 iecho->chksum = 0;
229 }
230#endif /* LWIP_CHECKSUM_CTRL_PER_NETIF */
231#else /* CHECKSUM_GEN_ICMP */
232 iecho->chksum = 0;
233#endif /* CHECKSUM_GEN_ICMP */
234
235 /* Set the correct TTL and recalculate the header checksum. */
236 IPH_TTL_SET(iphdr, ICMP_TTL);
237 IPH_CHKSUM_SET(iphdr, 0);
238#if CHECKSUM_GEN_IP
239 IF__NETIF_CHECKSUM_ENABLED(inp, NETIF_CHECKSUM_GEN_IP) {
240 IPH_CHKSUM_SET(iphdr, inet_chksum(iphdr, hlen));
241 }
242#endif /* CHECKSUM_GEN_IP */
243
244 ICMP_STATS_INC(icmp.xmit);
245 /* increase number of messages attempted to send */
246 MIB2_STATS_INC(mib2.icmpoutmsgs);
247 /* increase number of echo replies attempted to send */
248 MIB2_STATS_INC(mib2.icmpoutechoreps);
249
250 /* send an ICMP packet */
251 ret = ip4_output_if(p, src, LWIP_IP_HDRINCL,
252 ICMP_TTL, 0, IP_PROTO_ICMP, inp);
253 if (ret != ERR_OK) {
254 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: ip_output_if returned an error: %s\n", lwip_strerr(ret)));
255 }
256 }
257 break;
258 default:
259 if (type == ICMP_DUR) {
260 MIB2_STATS_INC(mib2.icmpindestunreachs);
261 } else if (type == ICMP_TE) {
262 MIB2_STATS_INC(mib2.icmpintimeexcds);
263 } else if (type == ICMP_PP) {
264 MIB2_STATS_INC(mib2.icmpinparmprobs);
265 } else if (type == ICMP_SQ) {
266 MIB2_STATS_INC(mib2.icmpinsrcquenchs);
267 } else if (type == ICMP_RD) {
268 MIB2_STATS_INC(mib2.icmpinredirects);
269 } else if (type == ICMP_TS) {
270 MIB2_STATS_INC(mib2.icmpintimestamps);
271 } else if (type == ICMP_TSR) {
272 MIB2_STATS_INC(mib2.icmpintimestampreps);
273 } else if (type == ICMP_AM) {
274 MIB2_STATS_INC(mib2.icmpinaddrmasks);
275 } else if (type == ICMP_AMR) {
276 MIB2_STATS_INC(mib2.icmpinaddrmaskreps);
277 }
278 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_input: ICMP type %"S16_F" code %"S16_F" not supported.\n",
279 (s16_t)type, (s16_t)code));
280 ICMP_STATS_INC(icmp.proterr);
281 ICMP_STATS_INC(icmp.drop);
282 }
283 pbuf_free(p);
284 return;
285lenerr:
286 pbuf_free(p);
287 ICMP_STATS_INC(icmp.lenerr);
288 MIB2_STATS_INC(mib2.icmpinerrors);
289 return;
290#if LWIP_ICMP_ECHO_CHECK_INPUT_PBUF_LEN || !LWIP_MULTICAST_PING || !LWIP_BROADCAST_PING
291icmperr:
292 pbuf_free(p);
293 ICMP_STATS_INC(icmp.err);
294 MIB2_STATS_INC(mib2.icmpinerrors);
295 return;
296#endif /* LWIP_ICMP_ECHO_CHECK_INPUT_PBUF_LEN || !LWIP_MULTICAST_PING || !LWIP_BROADCAST_PING */
297}

340static void
341icmp_send_response(struct pbuf *p, u8_t type, u8_t code)
342{
343 struct pbuf *q;
344 struct ip_hdr *iphdr;
345 struct icmp_hdr *icmphdr;
346 ip4_addr_t iphdr_src;
347 struct netif *netif;
348 u16_t response_pkt_len;
349
350 /* increase number of messages attempted to send */
351 MIB2_STATS_INC(mib2.icmpoutmsgs);
352
353 /* Keep IP header + up to 8 bytes */
354 response_pkt_len = IP_HLEN + ICMP_DEST_UNREACH_DATASIZE;
355 if (p->tot_len < response_pkt_len) {
356 response_pkt_len = p->tot_len;
357 }
358
359 /* ICMP header + part of original packet */
360 q = pbuf_alloc(PBUF_IP, sizeof(struct icmp_hdr) + response_pkt_len, PBUF_RAM);
361 if (q == NULL) {
362 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_send_response: failed to allocate pbuf for ICMP packet.\n"));
363 MIB2_STATS_INC(mib2.icmpouterrors);
364 return;
365 }
366 LWIP_ASSERT("check that first pbuf can hold icmp message",
367 (q->len >= (sizeof(struct icmp_hdr) + response_pkt_len)));
368
369 iphdr = (struct ip_hdr *)p->payload;
370 LWIP_DEBUGF(ICMP_DEBUG, ("icmp_send_response: Sending ICMP type %02X for packet from ", type));
371 ip4_addr_debug_print_val(ICMP_DEBUG, iphdr->src);
372 LWIP_DEBUGF(ICMP_DEBUG, (" to "));
373 ip4_addr_debug_print_val(ICMP_DEBUG, iphdr->dest);
374 LWIP_DEBUGF(ICMP_DEBUG, ("\n"));
375
376 icmphdr = (struct icmp_hdr *)q->payload;
377 icmphdr->type = type;
378 icmphdr->code = code;
379 icmphdr->data = 0;
380
381 /* copy fields from original packet */
382 pbuf_copy_partial_pbuf(q, p, response_pkt_len, sizeof(struct icmp_hdr));
383
384 ip4_addr_copy(iphdr_src, iphdr->src);
385#ifdef LWIP_HOOK_IP4_ROUTE_SRC
386 {
387 ip4_addr_t iphdr_dst;
388 ip4_addr_copy(iphdr_dst, iphdr->dest);
389 netif = ip4_route_src(&iphdr_dst, &iphdr_src);
390 }
391#else
392 netif = ip4_route(&iphdr_src);
393#endif
394 if (netif != NULL) {
395 /* calculate checksum */
396 icmphdr->chksum = 0;
397#if CHECKSUM_GEN_ICMP
398 IF__NETIF_CHECKSUM_ENABLED(netif, NETIF_CHECKSUM_GEN_ICMP) {
399 icmphdr->chksum = inet_chksum(icmphdr, q->len);
400 }
401#endif
402 ICMP_STATS_INC(icmp.xmit);
403 ip4_output_if(q, NULL, &iphdr_src, ICMP_TTL, 0, IP_PROTO_ICMP, netif);
404 }
405 pbuf_free(q);
406}
