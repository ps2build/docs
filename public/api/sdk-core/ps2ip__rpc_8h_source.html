183/* On-the-wire fd_set for the select RPC. The struct fd_set seen on EE
184 * (newlib, ~128 bytes) and IOP (lwIP, MEMP_NUM_NETCONN/8 bytes) compile
185 * units differs in size, which used to scramble select_pkt's layout
186 * across the SIF RPC boundary — IOP would only touch the first few
187 * bytes of EE's wider readset and never see writeset/exceptset at the
188 * right offsets. Using a fixed-size struct here keeps select_pkt
189 * byte-identical on both sides regardless of which struct fd_set is
190 * in scope at the include site. */
191typedef struct
192{
193 unsigned char fd_bits[(MEMP_NUM_NETCONN + 7) / 8];
194} ps2ip_rpc_fd_set;
195
196/* struct timeval differs across the boundary too — newlib's tv_sec/tv_usec
197 * are 64-bit on EE while the IOP defines them as 32-bit longs. Carry
198 * explicit 32-bit fields here so the layout is byte-identical on both
199 * sides; each end reconstructs its native struct timeval. */
200typedef struct

202 union

204 s32 maxfdp1;
205 s32 result;
206 };

210 void *readset_p; /* opaque: only NULL/non-NULL is meaningful across RPC */
211 void *writeset_p;
212 void *exceptset_p;
213 ps2ip_rpc_fd_set readset;
214 ps2ip_rpc_fd_set writeset;
215 ps2ip_rpc_fd_set exceptset;
216} select_pkt;
