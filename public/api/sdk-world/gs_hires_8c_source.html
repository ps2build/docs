133static void gsKit_setpass(GSGLOBAL *gsGlobal, const SPass * __pass, int yoff)
134{
135 u64 *p_data;
136 u64 *p_store;
137 int qsize = 8;
138
139 p_data = p_store = (u64*)gsKit_heap_alloc(gsGlobal, qsize, (qsize * 16), GIF_AD);
140
141 *p_data++ = GIF_TAG_AD(qsize);
142 *p_data++ = GIF_AD;
143
144 // Context 1
145 *p_data++ = GS_SETREG_SCISSOR_1(0, gsGlobal->Width - 1, 0, __pass->DrawBuffer->bufferline_end - __pass->DrawBuffer->bufferline_begin);
146 *p_data++ = GS_SCISSOR_1;
147 *p_data++ = GS_SETREG_FRAME_1(__pass->DrawBuffer->addr / 8192, gsGlobal->Width / 64, gsGlobal->PSM, 0);
148 *p_data++ = GS_FRAME_1;
149 *p_data++ = GS_SETREG_ZBUF_1(__pass->DepthBuffer->addr / 8192, gsGlobal->PSMZ, (gsGlobal->ZBuffering == GS_SETTING_ON) ? 0 : 1);
150 *p_data++ = GS_ZBUF_1;
151 *p_data++ = GS_SETREG_XYOFFSET_1(gsGlobal->OffsetX, yoff + gsGlobal->OffsetY + __pass->DrawBuffer->bufferline_begin*16);
152 *p_data++ = GS_XYOFFSET_1;
153
154 // Context 2
155 *p_data++ = GS_SETREG_SCISSOR_1(0, gsGlobal->Width - 1, 0, __pass->DrawBuffer->bufferline_end - __pass->DrawBuffer->bufferline_begin);
156 *p_data++ = GS_SCISSOR_2;
157 *p_data++ = GS_SETREG_FRAME_1(__pass->DrawBuffer->addr / 8192, gsGlobal->Width / 64, gsGlobal->PSM, 0);
158 *p_data++ = GS_FRAME_2;
159 *p_data++ = GS_SETREG_ZBUF_1(__pass->DepthBuffer->addr / 8192, gsGlobal->PSMZ, (gsGlobal->ZBuffering == GS_SETTING_ON) ? 0 : 1);
160 *p_data++ = GS_ZBUF_2;
161 *p_data++ = GS_SETREG_XYOFFSET_1(gsGlobal->OffsetX, yoff + gsGlobal->OffsetY + __pass->DrawBuffer->bufferline_begin*16);
162 *p_data++ = GS_XYOFFSET_2;
163}

165static void gsKit_setpass_background(GSGLOBAL *gsGlobal, const SPass * __pass, GSTEXTURE * tex)
166{
167 u16 width = gsGlobal->Width;
168 u16 height = __pass->DrawBuffer->bufferline_end - __pass->DrawBuffer->bufferline_begin + 1;
169 void * mem = &((u16*)tex->Mem)[__pass->DrawBuffer->bufferline_begin*width];

173}

175static void gsKit_create_pass(GSGLOBAL *gsGlobal, const SPass * __pass, GSQUEUE * Queue, int yoff)
176{
177 GSQUEUE * Queue_backup = gsGlobal->CurQueue;
178
179 // Create new queue (if not already created)
180 if (Queue->pool[0] == NULL)
181 gsKit_queue_init(gsGlobal, Queue, GS_PERSISTENT, 8*1024);
182 gsKit_queue_set(gsGlobal, Queue);
183 gsKit_queue_reset(Queue);
184
185 // Set the rendering context
186 gsKit_setpass(gsGlobal, __pass, yoff);
187
188 // Clear depth buffer (and background)
189 if ((gsGlobal->ZBuffering == GS_SETTING_ON) || (__pTexture == NULL))
190 gsKit_clear(gsGlobal, GS_SETREG_RGBAQ(0x00,0x00,0x00,0x00,0x00));
191
192 // Fill with background image
193 if (__pTexture != NULL)
194 gsKit_setpass_background(gsGlobal, __pass, __pTexture);
195
196 // Restore previous queue
197 gsKit_queue_set(gsGlobal, Queue_backup);
198}

200void _gsKit_create_passes(GSGLOBAL *gsGlobal)
201{
202 u32 iPass;
203
204 // Create draw queues to setup the rendering for each pass
205 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
206 for (iPass = 0; iPass < __iPassCount; iPass++) {
207 // ODD lines
208 gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].Queue, 0);
209 // EVEN lines, half a pixel DOWN
210 gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].QueueIF, 8);
211 }
212 }
213 else {
214 for (iPass = 0; iPass < __iPassCount; iPass++) {
215 gsKit_create_pass(gsGlobal, &__pass[iPass], &__pass[iPass].Queue, 0);
216 }
217 }
218}

220void gsKit_hires_sync(GSGLOBAL *gsGlobal)
221{
222 // Make sure the sema count == 0
223 while (PollSema(__sema_vsync_id) >= 0)
224 ;
225
226 // Wait for vsync
227 WaitSema(__sema_vsync_id);
228}

235// State tracking for rate limiting
236static u32 __last_flip_vsync_count = 0;
237
238void gsKit_hires_flip_ext(GSGLOBAL *gsGlobal, GSFLIP_MODE mode)
239{
240 u32 iY;
241 u32 vsync_end;
242
243 // Handle vsync waiting based on mode
244 switch (mode) {
245 case GSFLIP_DIRECT: vsync_end = __vsync_count; break;
246 case GSFLIP_VSYNC: vsync_end = __vsync_count + 1; break;
247 case GSFLIP_RATE_LIMIT_1: vsync_end = __last_flip_vsync_count + 1; break;
248 case GSFLIP_RATE_LIMIT_2: vsync_end = __last_flip_vsync_count + 2; break;
249 }
250
251 // Wait for required vsyncs
252 while (__vsync_count < vsync_end) {
253 WaitSema(__sema_vsync_id);
254 }
255
256 // Update last flip vsync count
257 __last_flip_vsync_count = __vsync_count;
258
259 // HACK: The start of the first displayed line
260 // this is needed by the hsync interrupt but must be
261 // updated every time StartYOffset changes
262 iY = gsGlobal->StartY + gsGlobal->StartYOffset + 4;
263 if (gsGlobal->Interlace == GS_INTERLACED)
264 iY /= 2;
265 __iYStart = iY;
266
267 // Toggle draw queue
268 WaitSema(__sema_queue_id);
269 dmaKit_wait_fast();
270 __iCurrentDrawQueue = (__iCurrentDrawQueue == 0) ? 1 : 0;
271 SignalSema(__sema_queue_id);
272
273 gsKit_queue_reset(&__DrawQueue[__iCurrentDrawQueue]);
274 gsKit_queue_set(gsGlobal, &__DrawQueue[__iCurrentDrawQueue]);
275}

277static void gsKit_texture_to_interlaced_frame(GSTEXTURE * tex)
278{
279 int y;
280 int bpp;
281 int size;
282 u16 *tex_new;
283 u16 *tex_old;
284
285 switch (tex->PSM)
286 {
287 case GS_PSM_CT16S: bpp = 2; break;
288 case GS_PSM_CT24: bpp = 3; break;
289 default: return; // not supported
290 };
291
292 size = tex->Width * tex->Height * bpp;
293 tex_new = (u16*)memalign(128, size);
294 tex_old = (u16*)tex->Mem;
295
296 for(y=0; y < tex->Height; y+=2) {
297 // Copy 0,2,4... to 0,1,2...
298 memcpy(&tex_new[(y/2)*tex->Width], &tex_old[y*tex->Width], tex->Width * bpp);
299 // Copy 1,3,5... to (height/2) + 0,1,2...
300 memcpy(&tex_new[((y+tex->Height)/2)*tex->Width], &tex_old[(y+1)*tex->Width], tex->Width * bpp);
301 }
302
303 free(tex->Mem);
304 tex->Mem = (void*)tex_new;
305}

307void gsKit_hires_prepare_bg(GSGLOBAL *gsGlobal, GSTEXTURE * tex)
308{
309 // Convert to same color format
310 if (tex->PSM != gsGlobal->PSM) {
311 if (gsGlobal->PSM == GS_PSM_CT16S)
312 gsKit_texture_to_psm16(tex);
313 else
314 return; // not supported
315 }
316
317 // Convert to interlaced (if needed)
318 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME))
319 gsKit_texture_to_interlaced_frame(tex);
320}

322int gsKit_hires_set_bg(GSGLOBAL *gsGlobal, GSTEXTURE * tex)
323{
324 // Validate width
325 if (tex->Width != gsGlobal->Width)
326 return -1;
327
328 // Validate height
329 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
330 if (tex->Height != gsGlobal->Height*2)
331 return -1;
332 }
333 else {
334 if (tex->Height != gsGlobal->Height)
335 return -1;
336 }
337
338 // Validate PSM
339 if (tex->PSM != gsGlobal->PSM)
340 return -1;
341
342 // Accept background texture
343 __pTexture = tex;
344 // (Re-)create pass queues
345 gsKit_hires_sync(gsGlobal);
346 WaitSema(__sema_queue_id);
347 dmaKit_wait_fast();
348 _gsKit_create_passes(gsGlobal);
349 SignalSema(__sema_queue_id);
350
351 return 0;
352}

354static int hsync_callback(int cause)
355{
356 u32 iPass;
357
358 if (*GS_CSR & (1<<2)) {
359 *GS_CSR = (1<<2);
360 __hsync_count++;
361 }
362
363 if (*GS_CSR & (1<<3)) {
364 *GS_CSR = (1<<3);
365 __vsync_count++;
366 __hsync_count=0;
367 }
368
369 for (iPass = 0; iPass < __iPassCount; iPass++) {
370 if (__hsync_count == (__iYStart + __pass[iPass].CRTCBuffer->scanline_begin)) {
371 if (iPass == 0) {
372 // We start rendering the first pass when the CRTC starts
373 // displaying the last pass of the previous field.
374 // So invert to get the current render field.
375 __iCurrentRenderField = (((GSREG*)GS_CSR)->FIELD == 1) ? 0 : 1;
376 }
377
378 __iCurrentPass = iPass;
379 iSignalSema(__sema_hsync_id);
380
381#ifdef BG_DEBUG
382 GS_SET_BGCOLOR((iPass+1)*63, (iPass+1)*63, (iPass+1)*63);
383#endif
384
385 break;
386 }
387 }
388
389 ExitHandler();
390 return 0;
391}

393static void gsKit_queue_send(GSGLOBAL *gsGlobal, GSQUEUE *Queue)
394{
395 if(Queue->tag_size == 0)
396 return;
397
398 *(u64 *)Queue->dma_tag = DMA_TAG(Queue->tag_size, 0, DMA_END, 0, 0, 0);
399
400 if(Queue->last_type != GIF_AD)
401 {
402 *(u64 *)Queue->last_tag = ((u64)Queue->same_obj | *(u64 *)Queue->last_tag);
403 }
404
405 dmaKit_wait_fast();
406 dmaKit_send_chain_ucab(DMA_CHANNEL_GIF, Queue->pool[Queue->dbuf]);
407}

409static void dma_to_gs_thread(void * data)
410{
411 GSGLOBAL *gsGlobal = (GSGLOBAL *)data;
412 u32 iQueue;
413
414 while(1)
415 {
416 // Make sure the sema count == 0
417 while (PollSema(__sema_hsync_id) >= 0)
418 ;
419
420 // Wait for CRTC
421 WaitSema(__sema_hsync_id);
422
423 WaitSema(__sema_queue_id);
424
425#ifdef BG_DEBUG
426 GS_SET_BGCOLOR(128,0,0);
427#endif
428
429 // Send current pass queue
430 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
431 if (__iCurrentRenderField == 1)
432 gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].Queue); // Odd
433 else
434 gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].QueueIF); // Even
435 }
436 else {
437 gsKit_queue_send(gsGlobal, &__pass[__iCurrentPass].Queue);
438 }
439
440 // Send current frame queue
441 iQueue = (__iCurrentDrawQueue == 0) ? 1 : 0;
442 gsKit_queue_send(gsGlobal, &__DrawQueue[iQueue]);
443
444#ifdef BG_DEBUG
445 GS_SET_BGCOLOR(0,128,0);
446#endif
447
448 SignalSema(__sema_queue_id);
449
450 // If this was the last pass, send vsync
451 if (__iCurrentPass == (__iPassCount-1))
452 SignalSema(__sema_vsync_id);
453 }

456void gsKit_hires_init_screen(GSGLOBAL *gsGlobal, int passCount)
457{
458 u8 ZBuffering_backup;
459 ee_sema_t sema;
460 u32 iPass;
461 u32 iBufferline;
462 u32 iPassHeight;
463 u32 iBpp;
464 u32 iHeightAlign;
465 ee_thread_t thread;
466
467 // Limit __iPassCount between 2 and 4:
468 // 2 Passes: Front buffer split in 2, used as front and back buffer. Separate depth buffer (half the size) needed.
469 // 3 Passes: Front buffer split in 3, used as front, back and depth buffer.
470 // 4 Passes: Front buffer split in 4, used as front, back and depth buffer + some temporary texture space (currently not used).
471 __iPassCount = passCount;
472 __iPassCount = __iPassCount < 2 ? 2 : __iPassCount;
473 __iPassCount = __iPassCount > 4 ? 4 : __iPassCount;
474
475 if (__iPassCount > 2) {
476 // Depth buffer uses the same buffers as the color buffer, so fix PSMZ
477 switch (gsGlobal->PSM) {
478 case GS_PSM_CT32: gsGlobal->PSMZ = GS_PSMZ_32; break;
479 case GS_PSM_CT24: gsGlobal->PSMZ = GS_PSMZ_32; break;
480 case GS_PSM_CT16: gsGlobal->PSMZ = GS_PSMZ_16; break;
481 case GS_PSM_CT16S: gsGlobal->PSMZ = GS_PSMZ_16S; break;
482 default:
483 printf("gsKit_hires: ERROR: invalid PSM\n");
484 return;
485 }
486 }
487
488#ifdef BG_DEBUG
489 if (gsGlobal->Width < 1024)
490 gsGlobal->Width -= 64;
491 else
492 gsGlobal->Width -= 128;
493#endif
494
495 // Bytes per pixel and height align for render buffers
496 if ((gsGlobal->PSM == GS_PSM_CT32) || (gsGlobal->PSM == GS_PSM_CT24)) {
497 iBpp = 4;
498 iHeightAlign = 32;
499 }
500 else {
501 iBpp = 2;
502 iHeightAlign = 64;
503 }
504
505 // The size of a pass must be a multiple of 32 or 64 lines depending on the PSM
506 iPassHeight = (gsGlobal->Height + (__iPassCount-1)) / __iPassCount;
507 iPassHeight = (iPassHeight + (iHeightAlign-1)) & ~(iHeightAlign-1);
508
509 // Double buffering is not used in multi pass rendering
510 // Force double buffering off
511 gsGlobal->DoubleBuffering = GS_SETTING_OFF;
512
513 // gsKit_init_screen must only allocate 1 front buffer
514 // so temporarily disable the depth buffer
515 ZBuffering_backup = gsGlobal->ZBuffering;
516 gsGlobal->ZBuffering = GS_SETTING_OFF;
517
518 // This initialize the display and allocates 1 front buffer
519 gsKit_init_screen(gsGlobal);
520
521 // We split the front buffer in different passes
522 // Each pass needs to be aligned, so the total alignment requirements become:
523 // - __iPassCount * 32 for 32/24 bit color buffers
524 // - __iPassCount * 64 for 16 bit color buffers
525 if (gsKit_texture_size(gsGlobal->Width, gsGlobal->Height, gsGlobal->PSM) < gsKit_texture_size(gsGlobal->Width, iPassHeight * __iPassCount, gsGlobal->PSM)) {
526 int iAlign = gsKit_texture_size(gsGlobal->Width, iPassHeight * __iPassCount, gsGlobal->PSM) - gsKit_texture_size(gsGlobal->Width, gsGlobal->Height, gsGlobal->PSM);
527 printf("gsKit_hires: Padding front buffer with %dKiB \n", iAlign/1024);
528 gsKit_vram_alloc(gsGlobal, iAlign, GSKIT_ALLOC_SYSBUFFER);
529 }
530
531 // Restore depth buffer setting
532 gsGlobal->ZBuffering = ZBuffering_backup;
533 if (gsGlobal->ZBuffering == GS_SETTING_ON)
534 gsKit_set_test(gsGlobal, GS_ZTEST_ON);
535
536 //
537 // The front buffer is split into the number of passes
538 // Fill the buffer array here
539 //
540 iBufferline = 0;
541 for (iPass = 0; iPass < __iPassCount; iPass++) {
542 // Starting address of buffer
543 __buffer[iPass].addr = iBufferline * gsGlobal->Width * iBpp;
544
545 // First and last line of buffer
546 __buffer[iPass].bufferline_begin = iBufferline;
547 iBufferline += iPassHeight;
548 if (iBufferline > gsGlobal->Height)
549 iBufferline = gsGlobal->Height;
550 __buffer[iPass].bufferline_end = iBufferline - 1;
551
552 // First and last line of buffer, measured in CRTC scanlines
553 __buffer[iPass].scanline_begin = __buffer[iPass].bufferline_begin;
554 __buffer[iPass].scanline_end = __buffer[iPass].bufferline_end;
555 if (gsGlobal->MagV > 0)
556 {
557 // Lines in the buffer are magnified (displayed more than once
558 __buffer[iPass].scanline_begin *= (gsGlobal->MagV+1);
559 __buffer[iPass].scanline_end *= (gsGlobal->MagV+1);
560 }
561 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FIELD))
562 {
563 // Only half of the lines in the buffer are read in interlaced field mode
564 __buffer[iPass].scanline_begin /= 2;
565 __buffer[iPass].scanline_end /= 2;
566 }
567 }
568
569 if ((gsGlobal->ZBuffering == GS_SETTING_ON) && (__iPassCount == 2)) {
570 // Depth buffer not inside front buffer, so allocate it
571 // We only need half the height of the depth buffer
572 gsGlobal->ZBuffer = gsKit_vram_alloc(gsGlobal, gsKit_texture_size(gsGlobal->Width, gsGlobal->Height/2, gsGlobal->PSMZ), GSKIT_ALLOC_SYSBUFFER);
573
574 // 3rd buffer is fixed to this depth buffer
575 __buffer[2].addr = gsGlobal->ZBuffer;
576 }
577
578 // first useable address for textures (after any potential Zbuffer allocation)
579 gsGlobal->TexturePointer = gsGlobal->CurrentPointer;
580
581 //
582 // What buffers need to be used in what pass? That's what the pass array is for.
583 // Fill the pass array
584 //
585 for (iPass = 0; iPass < __iPassCount; iPass++) {
586 int iPrevPass = (iPass == 0) ? (__iPassCount-1) : (iPass-1);
587
588 // We always draw top to bottom
589 __pass[iPass].DrawBuffer = &__buffer[iPass];
590 // The CRTC is right behind the draw buffer
591 __pass[iPass].CRTCBuffer = &__buffer[iPrevPass];
592
593 if (__iPassCount == 2) {
594 // Depth buffer fixed
595 __pass[iPass].DepthBuffer = &__buffer[2];
596 __pass[iPass].TextBuffer = NULL;
597 }
598 else if (__iPassCount == 3) {
599 int iNextPass = (iPass == (__iPassCount-1)) ? 0 : (iPass+1);
600
601 // Depth buffer inside front buffer
602 __pass[iPass].DepthBuffer = &__buffer[iNextPass];
603 __pass[iPass].TextBuffer = NULL;
604 }
605 else {
606 int iNextPass1 = (iPass == (__iPassCount-1)) ? 0 : (iPass+1);
607 int iNextPass2 = (iNextPass1 == (__iPassCount-1)) ? 0 : (iNextPass1+1);
608
609 // Depth and texture buffers inside front buffer
610 //
611 // Toggle the position of the buffers relative to the draw buffer
612 // so the texture buffer can stay in place twice as long
613 __pass[iPass].DepthBuffer = &__buffer[((iPass & 1) == 0) ? iNextPass1 : iNextPass2];
614 __pass[iPass].TextBuffer = &__buffer[((iPass & 1) == 0) ? iNextPass2 : iNextPass1];
615 }
616 }
617
618 // (Re-)create pass queues
619 _gsKit_create_passes(gsGlobal);
620
621 // Re-use the "per" queue from gsKit
622 __DrawQueue[0] = *gsGlobal->Per_Queue;
623 // Create a second identical queue for double buffering
624 gsKit_queue_init(gsGlobal, &__DrawQueue[1], GS_PERSISTENT, GS_RENDER_QUEUE_PER_POOLSIZE);
625 // Set active queue
626 gsKit_queue_set(gsGlobal, &__DrawQueue[__iCurrentDrawQueue]);
627
628 // Initialize semaphores
629 sema.init_count = 0;
630 sema.max_count = 1;
631 sema.option = 0;
632 if ((__sema_hsync_id = CreateSema(&sema)) < 0)
633 printf("gsKit_hires: ERROR: creating __sema_hsync_id\n");
634 sema.init_count = 0;
635 sema.max_count = 1;
636 sema.option = 0;
637 if ((__sema_vsync_id = CreateSema(&sema)) < 0)
638 printf("gsKit_hires: ERROR: creating __sema_vsync_id\n");
639 // Initialize mutex (also a semaphore)
640 sema.init_count = 1;
641 sema.max_count = 1;
642 sema.option = 0;

677

679 gsKit_remove_hsync_handler(__hsync_callback_id);
680

682

685
686 if (__sema_queue_id >= 0)
687 DeleteSema(__sema_queue_id);
688
689 if (__sema_vsync_id >= 0)
690 DeleteSema(__sema_vsync_id);
691
692 if (__sema_hsync_id >= 0)
693 DeleteSema(__sema_hsync_id);
694
695 dmaKit_wait_fast();
696 gsKit_queue_free(gsGlobal, &__DrawQueue[1]);
697
698 for (iPass = 0; iPass < __iPassCount; iPass++) {
699 gsKit_queue_free(gsGlobal, &__pass[iPass].Queue);
700 gsKit_queue_free(gsGlobal, &__pass[iPass].QueueIF);
701 }
702
703 gsKit_deinit_global(gsGlobal);
704}
