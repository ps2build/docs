115{
116 if (hooked_version != 0) {

118 DPRINTF(" while version 0x%x already hooked\n", hooked_version);
119 return;

121
122 // Only the newer sio2man libraries (with reset functions) are supported
123 if (lib->version > IRX_VER(1, 1)) {

195int sio2man_hook_init()
196{
197 iop_sema_t sema;
198 iop_library_t *lib;
199
200 //DPRINTF("%s\n", __FUNCTION__);
201
202 // Get ptr to intrmans internals
203 mmce_sio2_intrman_internals_ptr = GetIntrmanInternalData();
204 if (mmce_sio2_intrman_internals_ptr == NULL) {
205 DPRINTF("Failed to get intrman internals ptr\n");
206 return -1;
207 }
208
209 // Create semaphore for locking sio2man exclusively
210 sema.attr = 1;
211 sema.initial = 1;
212 sema.max = 1;
213 sema.option = 0;
214 lock_sema = CreateSema(&sema);
215 lock_sema2 = CreateSema(&sema);
216
217 // Hook into 'sio2man' now if it's already loaded
218 lib = ioplib_getByName("sio2man");
219 if (lib != NULL) {
220 // Get sio2man intr handler
221 sio2man_intr_handler_ptr = mmce_sio2_intrman_internals_ptr->interrupt_handler_table[17].handler; //TODO: mode set in lowest 2 bits
222 sio2man_intr_arg_ptr = mmce_sio2_intrman_internals_ptr->interrupt_handler_table[17].userdata;
223 DPRINTF("Got SIO2MAN intr handler @ 0x%p, arg @ 0x%p\n", sio2man_intr_handler_ptr, sio2man_intr_arg_ptr);
224
225 // Hook into sio2man
226 _sio2man_hook(lib);
227 ioplib_relinkExports(lib);
228
229 // 'sio2man' is not loaded yet
230 } else {
231 // Hook into 'loadcore' so we know when sio2man is loaded in the future
232 lib = ioplib_getByName("loadcore");
233 if (lib == NULL) {
234 DeleteSema(lock_sema);
235 DeleteSema(lock_sema2);
236 return -1;
237 }
238 pRegisterLibraryEntries = ioplib_hookExportEntry(lib, 6, hookRegisterLibraryEntries);
239
240 // Hook into intrman to get ptr to sio2man's intr handler in future
241 lib = ioplib_getByName("intrman");
242 if (lib == NULL) {
243 DeleteSema(lock_sema);
244 DeleteSema(lock_sema2);
245 return -1;
246 }
247 pRegisterIntrHandler = ioplib_hookExportEntry(lib, 4, hookRegisterIntrHandler);
248 }
249
250 return 0;
251}
