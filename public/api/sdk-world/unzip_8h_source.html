170extern unzFile ZEXPORT unzOpen(const char *path);
171extern unzFile ZEXPORT unzOpen64(const void *path);
172/*
173 Open a Zip file. path contain the full pathname (by example,
174 on a Windows XP computer "c:\\zlib\\zlib113.zip" or on an Unix computer
175 "zlib/zlib113.zip".
176 If the zipfile cannot be opened (file don't exist or in not valid), the
177 return value is NULL.
178 Else, the return value is a unzFile Handle, usable with other function
179 of this unzip package.
180 the "64" function take a const void* pointer, because the path is just the
181 value passed to the open64_file_func callback.
182 Under Windows, if UNICODE is defined, using fill_fopen64_filefunc, the path
183 is a pointer to a wide unicode string (LPCTSTR is LPCWSTR), so const char*
184 does not describe the reality
185*/
