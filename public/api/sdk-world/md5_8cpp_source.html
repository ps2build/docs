39 // mix functions for processBlock()
40 inline uint32_t f1(uint32_t b, uint32_t c, uint32_t d)
41 {
42 return d ^ (b & (c ^ d)); // original: f = (b & c) | ((~b) & d);
43 }
44
45 inline uint32_t f2(uint32_t b, uint32_t c, uint32_t d)
46 {
47 return c ^ (d & (b ^ c)); // original: f = (b & d) | (c & (~d));
48 }
49
50 inline uint32_t f3(uint32_t b, uint32_t c, uint32_t d)
51 {
52 return b ^ c ^ d;
53 }
54
55 inline uint32_t f4(uint32_t b, uint32_t c, uint32_t d)
56 {
57 return c ^ (b | ~d);
58 }
59
60 inline uint32_t rotate(uint32_t a, uint32_t c)
61 {
62 return (a << c) | (a >> (32 - c));
63 }
64
65#if defined(__BYTE_ORDER) && (__BYTE_ORDER != 0) && (__BYTE_ORDER == __BIG_ENDIAN)
66 inline uint32_t swap(uint32_t x)
67 {
68#if defined(__GNUC__) || defined(__clang__)
69 return __builtin_bswap32(x);
70#endif
71#ifdef MSC_VER
72 return _byteswap_ulong(x);
73#endif
74
75 return (x >> 24) |
76 ((x >> 8) & 0x0000FF00) |
77 ((x << 8) & 0x00FF0000) |
78 (x << 24);
79 }
80#endif
81}

212void MD5::add(const void* data, size_t numBytes)
213{
214 const uint8_t* current = (const uint8_t*) data;
215
216 if (m_bufferSize > 0)
217 {
218 while (numBytes > 0 && m_bufferSize < BlockSize)
219 {
220 m_buffer[m_bufferSize++] = *current++;
221 numBytes--;
222 }
223 }
224
225 // full buffer
226 if (m_bufferSize == BlockSize)
227 {
228 processBlock(m_buffer);
229 m_numBytes += BlockSize;
230 m_bufferSize = 0;
231 }
232
233 // no more data ?
234 if (numBytes == 0)
235 return;
236
237 // process full blocks
238 while (numBytes >= BlockSize)
239 {
240 processBlock(current);
241 current += BlockSize;
242 m_numBytes += BlockSize;
243 numBytes -= BlockSize;
244 }
245
246 // keep remaining bytes in buffer
247 while (numBytes > 0)
248 {
249 m_buffer[m_bufferSize++] = *current++;
250 numBytes--;
251 }
252}

256void MD5::processBuffer()
257{
258 // the input bytes are considered as bits strings, where the first bit is the most significant bit of the byte
259
260 // - append "1" bit to message
261 // - append "0" bits until message length in bit mod 512 is 448
262 // - append length as 64 bit integer
263
264 // number of bits
265 size_t paddedLength = m_bufferSize * 8;
266
267 // plus one bit set to 1 (always appended)
268 paddedLength++;
269
270 // number of bits must be (numBits % 512) = 448
271 size_t lower11Bits = paddedLength & 511;
272 if (lower11Bits <= 448)
273 paddedLength += 448 - lower11Bits;
274 else
275 paddedLength += 512 + 448 - lower11Bits;
276 // convert from bits to bytes
277 paddedLength /= 8;
278
279 // only needed if additional data flows over into a second block
280 unsigned char extra[BlockSize];
281
282 // append a "1" bit, 128 => binary 10000000
283 if (m_bufferSize < BlockSize)
284 m_buffer[m_bufferSize] = 128;
285 else
286 extra[0] = 128;
287
288 size_t i;
289 for (i = m_bufferSize + 1; i < BlockSize; i++)
290 m_buffer[i] = 0;
291 for (; i < paddedLength; i++)
292 extra[i - BlockSize] = 0;
293
294 // add message length in bits as 64 bit number
295 uint64_t msgBits = 8 * (m_numBytes + m_bufferSize);
296 // find right position
297 unsigned char* addLength;
298 if (paddedLength < BlockSize)
299 addLength = m_buffer + paddedLength;
300 else
301 addLength = extra + paddedLength - BlockSize;
302
303 // must be little endian
304 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
305 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
306 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
307 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
308 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
309 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
310 *addLength++ = msgBits & 0xFF; msgBits >>= 8;
311 *addLength++ = msgBits & 0xFF;
312
313 // process blocks
314 processBlock(m_buffer);
315 // flowed over into a second block ?
316 if (paddedLength > BlockSize)
317 processBlock(extra);
318}

322std::string MD5::getHash()
323{
324 // compute hash (as raw bytes)
325 unsigned char rawHash[HashBytes];
326 getHash(rawHash);
327
328 // convert to hex string
329 std::string result;
330 result.reserve(2 * HashBytes);
331 for (int i = 0; i < HashBytes; i++)
332 {
333 static const char dec2hex[16+1] = "0123456789abcdef";
334 result += dec2hex[(rawHash[i] >> 4) & 15];
335 result += dec2hex[ rawHash[i] & 15];
336 }
337
338 return result;
339}

343void MD5::getHash(unsigned char buffer[MD5::HashBytes])
344{
345 // save old hash if buffer is partially filled
346 uint32_t oldHash[HashValues];
347 for (int i = 0; i < HashValues; i++)
348 oldHash[i] = m_hash[i];
349
350 // process remaining bytes
351 processBuffer();
352
353 unsigned char* current = buffer;
354 for (int i = 0; i < HashValues; i++)
355 {
356 *current++ = m_hash[i] & 0xFF;
357 *current++ = (m_hash[i] >> 8) & 0xFF;
358 *current++ = (m_hash[i] >> 16) & 0xFF;
359 *current++ = (m_hash[i] >> 24) & 0xFF;
360
361 // restore old hash
362 m_hash[i] = oldHash[i];
363 }
364}
