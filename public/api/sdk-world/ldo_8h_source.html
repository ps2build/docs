21/*
22** Macro to check stack size and grow stack if needed. Parameters
23** 'pre'/'pos' allow the macro to preserve a pointer into the
24** stack across reallocations, doing the work only when needed.
25** It also allows the running of one GC step when the stack is
26** reallocated.
27** 'condmovestack' is used in heavy tests to force a stack reallocation
28** at every check.
29*/
30#define luaD_checkstackaux(L,n,pre,pos) \
31 if (l_unlikely(L->stack_last.p - L->top.p <= (n))) \
32 { pre; luaD_growstack(L, n, 1); pos; } \
33 else { condmovestack(L,pre,pos); }
34
35/* In general, 'pre'/'pos' are empty (nothing to save) */
36#define luaD_checkstack(L,n) luaD_checkstackaux(L,n,(void)0,(void)0)

40#define savestack(L,pt) (cast_charp(pt) - cast_charp(L->stack.p))
41#define restorestack(L,n) cast(StkId, cast_charp(L->stack.p) + (n))

44/* macro to check stack size, preserving 'p' */
45#define checkstackp(L,n,p) \
46 luaD_checkstackaux(L, n, \
47 ptrdiff_t t__ = savestack(L, p), /* save 'p' */ \
48 p = restorestack(L, t__)) /* 'pos' part: restore 'p' */

52#define checkstackGCp(L,n,p) \

55 luaC_checkGC(L), /* stack grow uses memory */ \
