23// Drawbuffer Heap Allocator, GSQUEUE version
24static inline void *_gsKit_heap_alloc(GSQUEUE *q, int qsize, int bsize, int type)
25{
26 void *p_heap;
27
28#ifdef GSKIT_DEBUG
29 if(((u32)q->pool_cur + bsize ) >= (u32)q->pool_max[q->dbuf])
30 {
31 printf("GSKIT: WARNING! HEAP OVERFLOW FOR RENDERQUEUE %p!!\n", q);
32 return NULL;
33 }
34#endif
35
36 if((q->tag_size + qsize) >= 65535)
37 {
38 *(u64 *)q->dma_tag = DMA_TAG(q->tag_size, 0, DMA_CNT, 0, 0, 0);
39 q->tag_size = 0;
40 q->dma_tag = q->pool_cur;
41 q->pool_cur = (u8*)q->pool_cur + 16;
42 }
43
44 if(type == GIF_AD || type != q->last_type || q->same_obj >= GS_GIF_BLOCKSIZE)
45 {
46 if(q->last_type != GIF_RESERVED && q->last_type != GIF_AD)
47 *(u64 *)q->last_tag = ((u64)q->same_obj | *(u64 *)q->last_tag);
48
49 qsize ++;
50 bsize += 16;
51 q->last_tag = q->pool_cur;
52 q->same_obj = 0;
53 }
54
55 q->same_obj++;
56 q->last_type = type;
57 q->tag_size += qsize;
58 p_heap = q->pool_cur;
59 q->pool_cur = (u8*)q->pool_cur + bsize;
60
61 return p_heap;
62}

70// Drawbuffer Heap Allocator (For Injected DMA_TAGs), GSQUEUE version
71static inline void *_gsKit_heap_alloc_dma(GSQUEUE *q, int qsize, int bsize)
72{
73 void *p_heap;
74
75#ifdef GSKIT_DEBUG
76 if(((u32)q->pool_cur + bsize ) >= (u32)q->pool_max[q->dbuf])
77 {
78 printf("GSKIT: WARNING! HEAP OVERFLOW FOR RENDERQUEUE %p!!\n", q);
79 return NULL;
80 }
81#endif
82
83 if(q->last_type != GIF_RESERVED && q->last_type != GIF_AD)
84 {
85 *(u64 *)q->last_tag = ((u64)q->same_obj | *(u64 *)q->last_tag);
86 }
87
88 *(u64 *)q->dma_tag = DMA_TAG(q->tag_size, 0, DMA_CNT, 0, 0, 0);
89 q->tag_size = 0;
90
91 q->last_type = GIF_RESERVED;
92 q->same_obj = 0;
93
94 p_heap = q->pool_cur;
95 q->pool_cur = (u8*)q->pool_cur + bsize;
96 q->dma_tag = q->pool_cur;
97 q->pool_cur = (u8*)q->pool_cur + 16;
98
99 return p_heap;
100}

108static inline int __gsKit_float_to_int_uv(float fuv, int imax)
109{
110 int iuv = (int)(fuv * 16.0f);
111
112 // Limit u/v to texture width/height
113
114 if (iuv < 0)
115 iuv = 0;
116
117 if (iuv > imax)
118 iuv = imax;
119
120 // Prevent overflow when using maximum size texture
121
122 if (iuv >= (1024 * 16))
123 iuv = (1024 * 16) - 1;
124
125 return iuv;
126}

138static inline int __gsKit_float_to_int_xy(float fxy, int offset)
139{
140 int ixy = (int)(fxy * 16.0f) + offset;

143

145 ixy = 0;
146
