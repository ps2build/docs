18/*
19** Collectable objects may have one of three colors: white, which means
20** the object is not marked; gray, which means the object is marked, but
21** its references may be not marked; and black, which means that the
22** object and all its references are marked. The main invariant of the
23** garbage collector, while marking objects, is that a black object can
24** never point to a white one. Moreover, any gray object must be in a
25** "gray list" (gray, grayagain, weak, allweak, ephemeron) so that it
26** can be visited again before finishing the collection cycle. (Open
27** upvalues are an exception to this rule.) These lists have no meaning
28** when the invariant is not being enforced (e.g., sweep phase).
29*/
30

50/*
51** macro to tell when main invariant (white objects cannot point to black
52** ones) must be kept. During a collection, the sweep
53** phase may break the invariant, as objects turned white may point to
54** still-black objects. The invariant is restored when sweep ends and
55** all objects are white again.
56*/
57
58#define keepinvariant(g) ((g)->gcstate <= GCSatomic)

109/* object age in generational mode */
110#define G_NEW 0 /* created in current cycle */
111#define G_SURVIVAL 1 /* created in previous cycle */
112#define G_OLD0 2 /* marked old by frw. barrier in this cycle */
113#define G_OLD1 3 /* first full cycle as old */
114#define G_OLD 4 /* really old object (not to be visited) */
115#define G_TOUCHED1 5 /* old object touched this cycle */
116#define G_TOUCHED2 6 /* old object touched in previous cycle */

118#define AGEBITS 7 /* all age bits (111) */

120#define getage(o) ((o)->marked & AGEBITS)
121#define setage(o,a) ((o)->marked = cast_byte(((o)->marked & (~AGEBITS)) | a))
122#define isold(o) (getage(o) > G_SURVIVAL)

165/*
166** Does one step of collection when debt becomes positive. 'pre'/'pos'
167** allows some adjustments to be done only when needed. macro
168** 'condchangemem' is used only for heavy tests (forcing a full
169** GC cycle on every opportunity)

171#define luaC_condGC(L,pre,pos) \
172 { if (G(L)->GCdebt > 0) { pre; luaC_step(L); pos;}; \
173 condchangemem(L,pre,pos); }
