53struct mad_header {
54 enum mad_layer layer; /* audio layer (1, 2, or 3) */
55 enum mad_mode mode; /* channel mode (see above) */
56 int mode_extension; /* additional mode info */
57 enum mad_emphasis emphasis; /* de-emphasis to use (see above) */
58
59 unsigned long bitrate; /* stream bitrate (bps) */
60 unsigned int samplerate; /* sampling frequency (Hz) */
61
62 unsigned short crc_check; /* frame CRC accumulator */
63 unsigned short crc_target; /* final target CRC checksum */
64
65 int flags; /* flags (see below) */
66 int private_bits; /* private bits (see below) */
67
68 mad_timer_t duration; /* audio playing time of frame */
69};

71struct mad_frame {

73
74 int options; /* decoding options (from stream) */
75
76 mad_fixed_t sbsample[2][36][32]; /* synthesis subband filter samples */
77 mad_fixed_t (*overlap)[2][32][18]; /* Layer III block overlap data */
78};
