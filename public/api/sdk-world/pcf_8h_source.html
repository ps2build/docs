183 /* macros for pcf font format */
184
185#define LSBFirst 0
186#define MSBFirst 1
187
188#define PCF_FILE_VERSION ( ( 'p' << 24 ) | \
189 ( 'c' << 16 ) | \
190 ( 'f' << 8 ) | 1 )
191#define PCF_FORMAT_MASK 0xFFFFFF00UL
192
193#define PCF_DEFAULT_FORMAT 0x00000000UL
194#define PCF_INKBOUNDS 0x00000200UL
195#define PCF_ACCEL_W_INKBOUNDS 0x00000100UL
196#define PCF_COMPRESSED_METRICS 0x00000100UL
197
198#define PCF_FORMAT_MATCH( a, b ) \
199 ( ( (a) & PCF_FORMAT_MASK ) == ( (b) & PCF_FORMAT_MASK ) )
200
201#define PCF_GLYPH_PAD_MASK ( 3 << 0 )
202#define PCF_BYTE_MASK ( 1 << 2 )
203#define PCF_BIT_MASK ( 1 << 3 )
204#define PCF_SCAN_UNIT_MASK ( 3 << 4 )
205
206#define PCF_BYTE_ORDER( f ) \
207 ( ( (f) & PCF_BYTE_MASK ) ? MSBFirst : LSBFirst )
208#define PCF_BIT_ORDER( f ) \
209 ( ( (f) & PCF_BIT_MASK ) ? MSBFirst : LSBFirst )
210#define PCF_GLYPH_PAD_INDEX( f ) \
211 ( (f) & PCF_GLYPH_PAD_MASK )
212#define PCF_GLYPH_PAD( f ) \
213 ( 1 << PCF_GLYPH_PAD_INDEX( f ) )
214#define PCF_SCAN_UNIT_INDEX( f ) \
215 ( ( (f) & PCF_SCAN_UNIT_MASK ) >> 4 )
216#define PCF_SCAN_UNIT( f ) \
217 ( 1 << PCF_SCAN_UNIT_INDEX( f ) )
218#define PCF_FORMAT_BITS( f ) \
219 ( (f) & ( PCF_GLYPH_PAD_MASK | \
220 PCF_BYTE_MASK | \
221 PCF_BIT_MASK | \
222 PCF_SCAN_UNIT_MASK ) )
223
224#define PCF_SIZE_TO_INDEX( s ) ( (s) == 4 ? 2 : (s) == 2 ? 1 : 0 )
225#define PCF_INDEX_TO_SIZE( b ) ( 1 << b )
226
227#define PCF_FORMAT( bit, byte, glyph, scan ) \
228 ( ( PCF_SIZE_TO_INDEX( scan ) << 4 ) | \
229 ( ( (bit) == MSBFirst ? 1 : 0 ) << 3 ) | \
230 ( ( (byte) == MSBFirst ? 1 : 0 ) << 2 ) | \
231 ( PCF_SIZE_TO_INDEX( glyph ) << 0 ) )
232
233#define PCF_PROPERTIES ( 1 << 0 )
234#define PCF_ACCELERATORS ( 1 << 1 )
235#define PCF_METRICS ( 1 << 2 )
236#define PCF_BITMAPS ( 1 << 3 )
237#define PCF_INK_METRICS ( 1 << 4 )
238#define PCF_BDF_ENCODINGS ( 1 << 5 )
239#define PCF_SWIDTHS ( 1 << 6 )
240#define PCF_GLYPH_NAMES ( 1 << 7 )
241#define PCF_BDF_ACCELERATORS ( 1 << 8 )
242
