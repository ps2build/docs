47
48 //Packet #1: Command, file descriptor, offset, and whence
49 mmce_sio2_lock();
50 res = mmce_sio2_tx_rx_pio(0xd, 0x16, wrbuf, rdbuf, &timeout_1s);
51 mmce_sio2_unlock();

58 DPRINTF("%s ERROR: Invalid response from card. Got 0x%x, Expected 0x%x\n", __func__, rdbuf[0x1], MMCE_REPLY_CONST);
59 return -1;
60 }
61
62 position = (s64)rdbuf[0xd] << 56;
63 position |= (s64)rdbuf[0xe] << 48;
64 position |= (s64)rdbuf[0xf] << 40;
65 position |= (s64)rdbuf[0x10] << 32;
66 position |= (s64)rdbuf[0x11] << 24;
67 position |= (s64)rdbuf[0x12] << 16;
68 position |= (s64)rdbuf[0x13] << 8;
69 position |= (s64)rdbuf[0x14];
70
71 DPRINTF("%s position %lli\n", __func__, position);
72

76int mmcedrv_read_sector(int fd, u32 sector, u32 count, void *buffer)
77{
78 int res;
79 int sectors_read = 0;
80
81 u8 wrbuf[0xB];
82 u8 rdbuf[0xB];
83
84 DPRINTF("%s fd: %i, starting sector: %i, count: %i\n", __func__, fd, sector, count);
85
86 wrbuf[0x0] = MMCE_ID; //Identifier
87 wrbuf[0x1] = MMCE_CMD_FS_READ_SECTOR; //Command
88 wrbuf[0x2] = MMCE_RESERVED; //Reserved
89 wrbuf[0x3] = fd; //File Descriptor
90
91 wrbuf[0x4] = (sector & 0x00FF0000) >> 16;
92 wrbuf[0x5] = (sector & 0x0000FF00) >> 8;
93 wrbuf[0x6] = (sector & 0x000000FF);
94
95 wrbuf[0x7] = (count & 0x00FF0000) >> 16;
96 wrbuf[0x8] = (count & 0x0000FF00) >> 8;
97 wrbuf[0x9] = (count & 0x000000FF);
98
99 wrbuf[0xA] = 0xff;
100
101 mmce_sio2_lock();
102
103 //Packet #1: Command, file descriptor, and size
104 res = mmce_sio2_tx_rx_pio(0xB, 0xB, wrbuf, rdbuf, &timeout_2s);
105 if (res == -1) {
106 DPRINTF("%s ERROR: P1 - Timedout waiting for /ACK\n", __func__);
107 mmce_sio2_unlock();
108 return -1;
109 }
110
111 if (rdbuf[0x1] != MMCE_REPLY_CONST) {
112 DPRINTF("%s ERROR: Invalid response from card. Got 0x%x, Expected 0x%x\n", __func__, rdbuf[0x1], MMCE_REPLY_CONST);
113 mmce_sio2_unlock();
114 return -1;
115 }
116
117 if (rdbuf[0x9] != 0x0) {
118 DPRINTF("%s ERROR: P1 - Got bad return value from card, res %i\n", __func__, rdbuf[0x9]);
119 mmce_sio2_unlock();
120 return -1;
121 }
122
123 //Packet #2 - n: Read data
124 res = mmce_sio2_rx_dma(buffer, (count * 2048));
125 if (res == -1) {
126 DPRINTF("%s ERROR: P2 - Timedout waiting for /ACK\n", __func__);
127 mmce_sio2_unlock();
128 return -1;
129 }
130
131 //Packet #n + 1: Sectors read
132 res = mmce_sio2_tx_rx_pio(0x0, 0x5, wrbuf, rdbuf, &timeout_1s);
133 if (res == -1) {
134 DPRINTF("%s ERROR: P3 - Timedout waiting for /ACK\n", __func__);
135 mmce_sio2_unlock();
136 return -1;
137 }
138
139 mmce_sio2_unlock();
140
141 sectors_read = rdbuf[0x1] << 16;
142 sectors_read |= rdbuf[0x2] << 8;
143 sectors_read |= rdbuf[0x3];
144
145 if (sectors_read != count) {
146 DPRINTF("%s ERROR: Sectors read: %i, expected: %i\n", __func__, sectors_read, count);
147 }
148
149 return sectors_read;
150}

373int __start(int argc, char *argv[])
374{
375 int rv;
376
377 DPRINTF("Multipurpose Memory Card Emulator Driver (MMCEDRV) v%d.%d by the MMCE team\n", MAJOR, MINOR);
378
379 //Install hooks
380 rv = mmce_sio2_init();
381 if (rv != 0) {
382 DPRINTF("mmce_sio2_init failed, rv %i\n", rv);
383 return MODULE_NO_RESIDENT_END;
384 }
385
386 //Register exports
387 if (RegisterLibraryEntries(&_exp_mmcedrv) != 0) {
388 DPRINTF("ERROR: library already registered\n");
389 return MODULE_NO_RESIDENT_END;
390 }
391
392 iop_library_t * lib_modload = ioplib_getByName("modload");
393 if (lib_modload != NULL) {
394 DPRINTF("modload 0x%x detected\n", lib_modload->version);
395 if (lib_modload->version > 0x102) //IOP is running a MODLOAD version which supports unloading IRX Modules
396 return MODULE_REMOVABLE_END; // and we do support getting unloaded...
397 } else {
398 DPRINTF("modload not detected! this is serious!\n");
399 }
400
401 return MODULE_RESIDENT_END;
402}
