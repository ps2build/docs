21/* INTERNAL, PRIVATE definition of a PNG.
22 *
23 * png_info is a modifiable description of a PNG datastream. The fields inside
24 * this structure are accessed through png_get_<CHUNK>() functions and modified
25 * using png_set_<CHUNK>() functions.
26 *
27 * Some functions in libpng do directly access members of png_info. However,
28 * this should be avoided. png_struct objects contain members which hold
29 * caches, sometimes optimised, of the values from png_info objects, and
30 * png_info is not passed to the functions which read and write image data.
31 */
32#ifndef PNGINFO_H
33#define PNGINFO_H
34
35struct png_info_def
36{
37 /* The following are necessary for every PNG file */
38 png_uint_32 width; /* width of image in pixels (from IHDR) */
39 png_uint_32 height; /* height of image in pixels (from IHDR) */
40 png_uint_32 valid; /* valid chunk data (see PNG_INFO_ below) */
41 size_t rowbytes; /* bytes needed to hold an untransformed row */
42 png_colorp palette; /* array of color values (valid & PNG_INFO_PLTE) */
43 png_uint_16 num_palette; /* number of color entries in "palette" (PLTE) */
44 png_uint_16 num_trans; /* number of transparent palette color (tRNS) */
45 png_byte bit_depth; /* 1, 2, 4, 8, or 16 bits/channel (from IHDR) */
46 png_byte color_type; /* see PNG_COLOR_TYPE_ below (from IHDR) */
47 /* The following three should have been named *_method not *_type */
48 png_byte compression_type; /* must be PNG_COMPRESSION_TYPE_BASE (IHDR) */
49 png_byte filter_type; /* must be PNG_FILTER_TYPE_BASE (from IHDR) */
50 png_byte interlace_type; /* One of PNG_INTERLACE_NONE, PNG_INTERLACE_ADAM7 */
51
52 /* The following are set by png_set_IHDR, called from the application on
53 * write, but the are never actually used by the write code.
54 */
55 png_byte channels; /* number of data channels per pixel (1, 2, 3, 4) */
56 png_byte pixel_depth; /* number of bits per pixel */
57 png_byte spare_byte; /* to align the data, and for future use */

59#ifdef PNG_READ_SUPPORTED

61 png_byte signature[8]; /* magic bytes read by libpng from start of file */
62#endif

64 /* The rest of the data is optional. If you are reading, check the

66 * are writing, set the valid field to those chunks you want written,

103#ifdef PNG_TEXT_SUPPORTED
104 /* The tEXt, and zTXt chunks contain human-readable textual data in
105 * uncompressed, compressed, and optionally compressed forms, respectively.
106 * The data in "text" is an array of pointers to uncompressed,
107 * null-terminated C strings. Each chunk has a keyword that describes the
108 * textual data contained in that chunk. Keywords are not required to be
109 * unique, and the text string may be empty. Any number of text chunks may
110 * be in an image.
111 */
112 int num_text; /* number of comments read or comments to write */
113 int max_text; /* current size of text array */
114 png_textp text; /* array of comments read or comments to write */
115#endif /* TEXT */

121 png_time mod_time;

124#ifdef PNG_sBIT_SUPPORTED
125 /* The sBIT chunk specifies the number of significant high-order bits
126 * in the pixel data. Values are in the range [1, bit_depth], and are
127 * only specified for the channels in the pixel data. The contents of
128 * the low-order bits is not specified. Data is valid if
129 * (valid & PNG_INFO_sBIT) is non-zero.
130 */
131 png_color_8 sig_bit; /* significant bits in color channels */
132#endif

134#if defined(PNG_tRNS_SUPPORTED) || defined(PNG_READ_EXPAND_SUPPORTED) || \
135defined(PNG_READ_BACKGROUND_SUPPORTED)
136 /* The tRNS chunk supplies transparency data for paletted images and
137 * other image types that don't need a full alpha channel. There are
138 * "num_trans" transparency values for a paletted image, stored in the
139 * same order as the palette colors, starting from index 0. Values
140 * for the data are in the range [0, 255], ranging from fully transparent
141 * to fully opaque, respectively. For non-paletted images, there is a
142 * single color specified that should be treated as fully transparent.
143 * Data is valid if (valid & PNG_INFO_tRNS) is non-zero.
144 */
145 png_bytep trans_alpha; /* alpha values for paletted image */
146 png_color_16 trans_color; /* transparent color for non-palette image */
147#endif
148
149#if defined(PNG_bKGD_SUPPORTED) || defined(PNG_READ_BACKGROUND_SUPPORTED)
150 /* The bKGD chunk gives the suggested image background color if the
151 * display program does not have its own background color and the image
152 * is needs to composited onto a background before display. The colors
153 * in "background" are normally in the same color space/depth as the
154 * pixel data. Data is valid if (valid & PNG_INFO_bKGD) is non-zero.
155 */
156 png_color_16 background;
157#endif

159#ifdef PNG_oFFs_SUPPORTED

161 * and downwards from the top-left corner of the display, page, or other
162 * application-specific co-ordinate space. See the PNG_OFFSET_ defines
163 * below for the unit types. Valid if (valid & PNG_INFO_oFFs) non-zero.
164 */
165 png_int_32 x_offset; /* x offset on page */
166 png_int_32 y_offset; /* y offset on page */
167 png_byte offset_unit_type; /* offset units type */
168#endif

174 */
175 png_uint_32 x_pixels_per_unit; /* horizontal pixel density */
176 png_uint_32 y_pixels_per_unit; /* vertical pixel density */
177 png_byte phys_unit_type; /* resolution type (see PNG_RESOLUTION_ below) */
178#endif

182 png_bytep exif;

185#ifdef PNG_hIST_SUPPORTED
186 /* The hIST chunk contains the relative frequency or importance of the
187 * various palette entries, so that a viewer can intelligently select a
188 * reduced-color palette, if required. Data is an array of "num_palette"
189 * values in the range [0,65535]. Data valid if (valid & PNG_INFO_hIST)
190 * is non-zero.
191 */
192 png_uint_16p hist;
193#endif

195#ifdef PNG_pCAL_SUPPORTED
196 /* The pCAL chunk describes a transformation between the stored pixel
197 * values and original physical data values used to create the image.
198 * The integer range [0, 2^bit_depth - 1] maps to the floating-point
199 * range given by [pcal_X0, pcal_X1], and are further transformed by a
200 * (possibly non-linear) transformation function given by "pcal_type"
201 * and "pcal_params" into "pcal_units". Please see the PNG_EQUATION_
202 * defines below, and the PNG-Group's PNG extensions document for a
203 * complete description of the transformations and how they should be
204 * implemented, and for a description of the ASCII parameter strings.
205 * Data values are valid if (valid & PNG_INFO_pCAL) non-zero.
206 */
207 png_charp pcal_purpose; /* pCAL chunk description string */
208 png_int_32 pcal_X0; /* minimum value */
209 png_int_32 pcal_X1; /* maximum value */
210 png_charp pcal_units; /* Latin-1 string giving physical units */
211 png_charpp pcal_params; /* ASCII strings containing parameter values */
212 png_byte pcal_type; /* equation type (see PNG_EQUATION_ below) */
213 png_byte pcal_nparams; /* number of parameters given in pcal_params */
214#endif
