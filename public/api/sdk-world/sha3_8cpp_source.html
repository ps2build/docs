94void SHA3::processBlock(const void* data)
95{
96#if defined(__BYTE_ORDER) && (__BYTE_ORDER != 0) && (__BYTE_ORDER == __BIG_ENDIAN)
97#define LITTLEENDIAN(x) swap(x)
98#else
99#define LITTLEENDIAN(x) (x)
100#endif
101
102 const uint64_t* data64 = (const uint64_t*) data;
103 // mix data into state
104 for (unsigned int i = 0; i < m_blockSize / 8; i++)
105 m_hash[i] ^= LITTLEENDIAN(data64[i]);
106
107 // re-compute state
108 for (unsigned int round = 0; round < Rounds; round++)
109 {
110 // Theta
111 uint64_t coefficients[5];
112 for (unsigned int i = 0; i < 5; i++)
113 coefficients[i] = m_hash[i] ^ m_hash[i + 5] ^ m_hash[i + 10] ^ m_hash[i + 15] ^ m_hash[i + 20];
114
115 for (unsigned int i = 0; i < 5; i++)
116 {
117 uint64_t one = coefficients[mod5(i + 4)] ^ rotateLeft(coefficients[mod5(i + 1)], 1);
118 m_hash[i ] ^= one;
119 m_hash[i + 5] ^= one;
120 m_hash[i + 10] ^= one;
121 m_hash[i + 15] ^= one;
122 m_hash[i + 20] ^= one;
123 }
124
125 // temporary
126 uint64_t one;
127
128 // Rho Pi
129 uint64_t last = m_hash[1];
130 one = m_hash[10]; m_hash[10] = rotateLeft(last, 1); last = one;
131 one = m_hash[ 7]; m_hash[ 7] = rotateLeft(last, 3); last = one;
132 one = m_hash[11]; m_hash[11] = rotateLeft(last, 6); last = one;
133 one = m_hash[17]; m_hash[17] = rotateLeft(last, 10); last = one;
134 one = m_hash[18]; m_hash[18] = rotateLeft(last, 15); last = one;
135 one = m_hash[ 3]; m_hash[ 3] = rotateLeft(last, 21); last = one;
136 one = m_hash[ 5]; m_hash[ 5] = rotateLeft(last, 28); last = one;
137 one = m_hash[16]; m_hash[16] = rotateLeft(last, 36); last = one;
138 one = m_hash[ 8]; m_hash[ 8] = rotateLeft(last, 45); last = one;
139 one = m_hash[21]; m_hash[21] = rotateLeft(last, 55); last = one;
140 one = m_hash[24]; m_hash[24] = rotateLeft(last, 2); last = one;
141 one = m_hash[ 4]; m_hash[ 4] = rotateLeft(last, 14); last = one;
142 one = m_hash[15]; m_hash[15] = rotateLeft(last, 27); last = one;
143 one = m_hash[23]; m_hash[23] = rotateLeft(last, 41); last = one;
144 one = m_hash[19]; m_hash[19] = rotateLeft(last, 56); last = one;
145 one = m_hash[13]; m_hash[13] = rotateLeft(last, 8); last = one;
146 one = m_hash[12]; m_hash[12] = rotateLeft(last, 25); last = one;
147 one = m_hash[ 2]; m_hash[ 2] = rotateLeft(last, 43); last = one;
148 one = m_hash[20]; m_hash[20] = rotateLeft(last, 62); last = one;
149 one = m_hash[14]; m_hash[14] = rotateLeft(last, 18); last = one;
150 one = m_hash[22]; m_hash[22] = rotateLeft(last, 39); last = one;
151 one = m_hash[ 9]; m_hash[ 9] = rotateLeft(last, 61); last = one;
152 one = m_hash[ 6]; m_hash[ 6] = rotateLeft(last, 20); last = one;
153 m_hash[ 1] = rotateLeft(last, 44);
154
155 // Chi
156 for (unsigned int j = 0; j < StateSize; j += 5)
157 {
158 // temporaries
159 uint64_t one = m_hash[j];
160 uint64_t two = m_hash[j + 1];
161
162 m_hash[j] ^= m_hash[j + 2] & ~two;
163 m_hash[j + 1] ^= m_hash[j + 3] & ~m_hash[j + 2];
164 m_hash[j + 2] ^= m_hash[j + 4] & ~m_hash[j + 3];
165 m_hash[j + 3] ^= one & ~m_hash[j + 4];
166 m_hash[j + 4] ^= two & ~one;
167 }
168
169 // Iota
170 m_hash[0] ^= XorMasks[round];
171 }
172}

176void SHA3::add(const void* data, size_t numBytes)
177{
178 const uint8_t* current = (const uint8_t*) data;
179
180 // copy data to buffer
181 if (m_bufferSize > 0)
182 {
183 while (numBytes > 0 && m_bufferSize < m_blockSize)
184 {
185 m_buffer[m_bufferSize++] = *current++;
186 numBytes--;
187 }
188 }
189
190 // full buffer
191 if (m_bufferSize == m_blockSize)
192 {
193 processBlock((void*)m_buffer);
194 m_numBytes += m_blockSize;
195 m_bufferSize = 0;
196 }
197
198 // no more data ?
199 if (numBytes == 0)
200 return;
201
202 // process full blocks
203 while (numBytes >= m_blockSize)
204 {
205 processBlock(current);
206 current += m_blockSize;
207 m_numBytes += m_blockSize;
208 numBytes -= m_blockSize;
209 }
210
211 // keep remaining bytes in buffer
212 while (numBytes > 0)
213 {
214 m_buffer[m_bufferSize++] = *current++;
215 numBytes--;
216 }
217}

221void SHA3::processBuffer()
222{
223 // add padding
224 size_t offset = m_bufferSize;
225 // add a "1" byte
226 m_buffer[offset++] = 0x06;
227 // fill with zeros
228 while (offset < m_blockSize)
229 m_buffer[offset++] = 0;
230
231 // and add a single set bit
232 m_buffer[offset - 1] |= 0x80;
233
234 processBlock(m_buffer);
235}

239std::string SHA3::getHash()
240{
241 // save hash state
242 uint64_t oldHash[StateSize];
243 for (unsigned int i = 0; i < StateSize; i++)
244 oldHash[i] = m_hash[i];
245
246 // process remaining bytes
247 processBuffer();
248
249 // convert hash to string
250 static const char dec2hex[16 + 1] = "0123456789abcdef";
251
252 // number of significant elements in hash (uint64_t)
253 unsigned int hashLength = m_bits / 64;
254
255 std::string result;
256 result.reserve(m_bits / 4);
257 for (unsigned int i = 0; i < hashLength; i++)
258 for (unsigned int j = 0; j < 8; j++) // 64 bits => 8 bytes
259 {
260 // convert a byte to hex
261 unsigned char oneByte = (unsigned char) (m_hash[i] >> (8 * j));
262 result += dec2hex[oneByte >> 4];
263 result += dec2hex[oneByte & 15];
264 }
265
266 // SHA3-224's last entry in m_hash provides only 32 bits instead of 64 bits
267 unsigned int remainder = m_bits - hashLength * 64;
268 unsigned int processed = 0;
269 while (processed < remainder)
270 {
271 // convert a byte to hex
272 unsigned char oneByte = (unsigned char) (m_hash[hashLength] >> processed);
273 result += dec2hex[oneByte >> 4];
274 result += dec2hex[oneByte & 15];
275
276 processed += 8;
277 }
278
279 // restore state
280 for (unsigned int i = 0; i < StateSize; i++)
281 m_hash[i] = oldHash[i];
282
283 return result;
284}
