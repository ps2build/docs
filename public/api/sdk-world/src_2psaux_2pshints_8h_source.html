55 /*
56 * A HintMask object stores a bit mask that specifies which hints in the
57 * charstring are active at a given time. Hints in CFF must be declared
58 * at the start, before any drawing operators, with horizontal hints
59 * preceding vertical hints. The HintMask is ordered the same way, with
60 * horizontal hints immediately followed by vertical hints. Clients are
61 * responsible for knowing how many of each type are present.
62 *
63 * The maximum total number of hints is 96, as specified by the CFF
64 * specification.
65 *
66 * A HintMask is built 0 or more times while interpreting a charstring, by
67 * the HintMask operator. There is only one HintMask, but it is built or
68 * rebuilt each time there is a hint substitution (HintMask operator) in
69 * the charstring. A default HintMask with all bits set is built if there
70 * has been no HintMask operator prior to the first drawing operator.
71 *
72 */
73

79 FT_Bool isNew;

81 size_t bitCount;
82 size_t byteCount;
83
84 FT_Byte mask[( CF2_MAX_HINTS + 7 ) / 8];
85
86 } CF2_HintMaskRec, *CF2_HintMask;

102 /*
103 * A HintMap object stores a piecewise linear function for mapping
104 * y-coordinates from character space to device space, providing
105 * appropriate pixel alignment to stem edges.
106 *
107 * The map is implemented as an array of `CF2_Hint' elements, each
108 * representing an edge. When edges are paired, as from stem hints, the
109 * bottom edge must immediately precede the top edge in the array.
110 * Element character space AND device space positions must both increase
111 * monotonically in the array. `CF2_Hint' elements are also used as
112 * parameters to `cf2_blues_capture'.
113 *
114 * The `cf2_hintmap_build' method must be called before any drawing
115 * operation (beginning with a Move operator) and at each hint
116 * substitution (HintMask operator).
117 *
118 * The `cf2_hintmap_map' method is called to transform y-coordinates at
119 * each drawing operation (move, line, curve).
120 *
121 */
122
