103/* Compile time options.
104 * =====================
105 * In a multi-arch build the compiler may compile the code several times for the
106 * same object module, producing different binaries for different architectures.
107 * When this happens configure-time setting of the target host options cannot be
108 * done and this interferes with the handling of the ARM NEON optimizations, and
109 * possibly other similar optimizations. Put additional tests here; in general
110 * this is needed when the same option can be changed at both compile time and
111 * run time depending on the target OS (i.e. iOS vs Android.)
112 *
113 * NOTE: symbol prefixing does not pass $(CFLAGS) to the preprocessor, because
114 * this is not possible with certain compilers (Oracle SUN OS CC), as a result
115 * it is necessary to ensure that all extern functions that *might* be used
116 * regardless of $(CFLAGS) get declared in this file. The test on __ARM_NEON__
117 * below is one example of this behavior because it is controlled by the
118 * presence or not of -mfpu=neon on the GCC command line, it is possible to do
119 * this in $(CC), e.g. "CC=gcc -mfpu=neon", but people who build libpng rarely
120 * do this.
121 */
122#ifndef PNG_ARM_NEON_OPT
123 /* ARM NEON optimizations are being controlled by the compiler settings,
124 * typically the target FPU. If the FPU has been set to NEON (-mfpu=neon
125 * with GCC) then the compiler will define __ARM_NEON__ and we can rely
126 * unconditionally on NEON instructions not crashing, otherwise we must
127 * disable use of NEON instructions.
128 *
129 * NOTE: at present these optimizations depend on 'ALIGNED_MEMORY', so they
130 * can only be turned on automatically if that is supported too. If
131 * PNG_ARM_NEON_OPT is set in CPPFLAGS (to >0) then arm/arm_init.c will fail
132 * to compile with an appropriate #error if ALIGNED_MEMORY has been turned
133 * off.
134 *
135 * Note that gcc-4.9 defines __ARM_NEON instead of the deprecated
136 * __ARM_NEON__, so we check both variants.
137 *
138 * To disable ARM_NEON optimizations entirely, and skip compiling the
139 * associated assembler code, pass --enable-arm-neon=no to configure
140 * or put -DPNG_ARM_NEON_OPT=0 in CPPFLAGS.
141 */
142# if (defined(__ARM_NEON__) || defined(__ARM_NEON)) && \
143 defined(PNG_ALIGNED_MEMORY_SUPPORTED)
144# define PNG_ARM_NEON_OPT 2
145# else

147# endif
148#endif

1399/* Combine a row of data, dealing with alpha, etc. if requested. 'row' is an
1400 * array of png_ptr->width pixels. If the image is not interlaced or this
1401 * is the final pass this just does a memcpy, otherwise the "display" flag
1402 * is used to determine whether to copy pixels that are not in the current pass.
1403 *
1404 * Because 'png_do_read_interlace' (below) replicates pixels this allows this
1405 * function to achieve the documented 'blocky' appearance during interlaced read
1406 * if display is 1 and the 'sparkle' appearance, where existing pixels in 'row'
1407 * are not changed if they are not in the current pass, when display is 0.
1408 *
1409 * 'display' must be 0 or 1, otherwise the memcpy will be done regardless.
1410 *
1411 * The API always reads from the png_struct row buffer and always assumes that
1412 * it is full width (png_do_read_interlace has already been called.)
1413 *
1414 * This function is only ever used to write to row buffers provided by the
1415 * caller of the relevant libpng API and the row must have already been
1416 * transformed by the read transformations.
1417 *
1418 * The PNG_USE_COMPILE_TIME_MASKS option causes generation of pre-computed
1419 * bitmasks for use within the code, otherwise runtime generated masks are used.
1420 * The default is compile time masks.
1421 */

1423# define PNG_USE_COMPILE_TIME_MASKS 1

1906#if defined(PNG_sCAL_SUPPORTED) || defined(PNG_pCAL_SUPPORTED)
1907/* An internal API to validate the format of a floating point number.
1908 * The result is the index of the next character. If the number is
1909 * not valid it will be the index of a character in the supposed number.
1910 *
1911 * The format of a number is defined in the PNG extensions specification
1912 * and this API is strictly conformant to that spec, not anyone elses!
1913 *
1914 * The format as a regular expression is:
1915 *
1916 * [+-]?[0-9]+.?([Ee][+-]?[0-9]+)?
1917 *
1918 * or:
1919 *
1920 * [+-]?.[0-9]+(.[0-9]+)?([Ee][+-]?[0-9]+)?
1921 *
1922 * The complexity is that either integer or fraction must be present and the
1923 * fraction is permitted to have no digits only if the integer is present.
1924 *
1925 * NOTE: The dangling E problem.
1926 * There is a PNG valid floating point number in the following:
1927 *
1928 * PNG floating point numbers are not greedy.
1929 *
1930 * Working this out requires *TWO* character lookahead (because of the
1931 * sign), the parser does not do this - it will fail at the 'r' - this
1932 * doesn't matter for PNG sCAL chunk values, but it requires more care
1933 * if the value were ever to be embedded in something more complex. Use
1934 * ANSI-C strtod if you need the lookahead.
1935 */
1936/* State table for the parser. */
1937#define PNG_FP_INTEGER 0 /* before or in integer */
1938#define PNG_FP_FRACTION 1 /* before or in fraction */
1939#define PNG_FP_EXPONENT 2 /* before or in exponent */
1940#define PNG_FP_STATE 3 /* mask for the above */
1941#define PNG_FP_SAW_SIGN 4 /* Saw +/- in current state */
1942#define PNG_FP_SAW_DIGIT 8 /* Saw a digit in current state */
1943#define PNG_FP_SAW_DOT 16 /* Saw a dot in current state */
1944#define PNG_FP_SAW_E 32 /* Saw an E (or e) in current state */
1945#define PNG_FP_SAW_ANY 60 /* Saw any of the above 4 */
1946
1947/* These three values don't affect the parser. They are set but not used.
1948 */
1949#define PNG_FP_WAS_VALID 64 /* Preceding substring is a valid fp number */
1950#define PNG_FP_NEGATIVE 128 /* A negative number, including "-0" */
1951#define PNG_FP_NONZERO 256 /* A non-zero value */
1952#define PNG_FP_STICKY 448 /* The above three flags */
1953
1954/* This is available for the caller to store in 'state' if required. Do not
1955 * call the parser after setting it (the parser sometimes clears it.)
1956 */
1957#define PNG_FP_INVALID 512 /* Available for callers as a distinct value */
1958
1959/* Result codes for the parser (boolean - true means ok, false means
1960 * not ok yet.)
1961 */
1962#define PNG_FP_MAYBE 0 /* The number may be valid in the future */
1963#define PNG_FP_OK 1 /* The number is valid */
1964
1965/* Tests on the sticky non-zero and negative flags. To pass these checks
1966 * the state must also indicate that the whole number is valid - this is
1967 * achieved by testing PNG_FP_SAW_DIGIT (see the implementation for why this
1968 * is equivalent to PNG_FP_OK above.)
1969 */
1970#define PNG_FP_NZ_MASK (PNG_FP_SAW_DIGIT | PNG_FP_NEGATIVE | PNG_FP_NONZERO)
1971 /* NZ_MASK: the string is valid and a non-zero negative value */
1972#define PNG_FP_Z_MASK (PNG_FP_SAW_DIGIT | PNG_FP_NONZERO)
1973 /* Z MASK: the string is valid and a non-zero value. */
1974 /* PNG_FP_SAW_DIGIT: the string is valid. */
1975#define PNG_FP_IS_ZERO(state) (((state) & PNG_FP_Z_MASK) == PNG_FP_SAW_DIGIT)
1976#define PNG_FP_IS_POSITIVE(state) (((state) & PNG_FP_NZ_MASK) == PNG_FP_Z_MASK)
1977#define PNG_FP_IS_NEGATIVE(state) (((state) & PNG_FP_NZ_MASK) == PNG_FP_NZ_MASK)
1978
1979/* The actual parser. This can be called repeatedly. It updates
1980 * the index into the string and the state variable (which must
1981 * be initialized to 0). It returns a result code, as above. There
1982 * is no point calling the parser any more if it fails to advance to
1983 * the end of the string - it is stuck on an invalid character (or
