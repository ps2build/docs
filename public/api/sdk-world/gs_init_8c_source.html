28short int gsKit_check_rom(void)
29{
30 static int default_signal = -1;
31 char romname[15];
32
33 if(default_signal < 0)
34 {
35 _io_driver driver = posixIODriver;
36 GetRomNameWithIODriver((char *)romname, &driver);
37 romname[14] = '\0';
38
39 //ROMVER string format: VVVVRTYYYYMMDD
40 default_signal = (romname[4] == 'E') ? GS_MODE_PAL : GS_MODE_NTSC;
41 }
42
43 return default_signal;
44}

46void gsKit_set_display_offset(GSGLOBAL *gsGlobal, int x, int y)
47{
48 gsGlobal->StartXOffset = x;
49 gsGlobal->StartYOffset = y;
50
51 if (gsGlobal->Interlace == GS_INTERLACED) {
52 // Do not change odd/even start position in interlaced mode
53 gsGlobal->StartYOffset &= ~1;
54 }
55
56 GS_SET_DISPLAY1(
57 gsGlobal->StartX + gsGlobal->StartXOffset, // X position in the display area (in VCK unit
58 gsGlobal->StartY + gsGlobal->StartYOffset, // Y position in the display area (in Raster u
59 gsGlobal->MagH, // Horizontal Magnification
60 gsGlobal->MagV, // Vertical Magnification
61 gsGlobal->DW - 1, // Display area width
62 gsGlobal->DH - 1); // Display area height
63
64 GS_SET_DISPLAY2(
65 gsGlobal->StartX + gsGlobal->StartXOffset, // X position in the display area (in VCK units)
66 gsGlobal->StartY + gsGlobal->StartYOffset, // Y position in the display area (in Raster units)
67 gsGlobal->MagH, // Horizontal Magnification
68 gsGlobal->MagV, // Vertical Magnification
69 gsGlobal->DW - 1, // Display area width
70 gsGlobal->DH - 1); // Display area height
71}

195 gsGlobal->DH = 1080;
196 break;
197 }
198
199 if((gsGlobal->Mode == GS_MODE_NTSC || gsGlobal->Mode == GS_MODE_PAL) && (gsGlobal->Interlace == GS_NONINTERLACED)) {
200 // NTSC 240p instead of 480i
201 // PAL 288p instead of 576i
202 gsGlobal->StartY /= 2;
203 gsGlobal->DH /= 2;
204 }
205
206 gsGlobal->MagH = (gsGlobal->DW / gsGlobal->Width) - 1; // gsGlobal->DW should be a multiple of the screen width
207 gsGlobal->MagV = (gsGlobal->DH / gsGlobal->Height) - 1; // gsGlobal->DH should be a multiple of the screen height
208
209 // For other video modes, other than NTSC and PAL: if this model supports the GetGsDxDy syscall, get the board-specific offsets.
210 if(gsGlobal->Mode != GS_MODE_NTSC && gsGlobal->Mode != GS_MODE_PAL)
211 {
212 if(GetSyscallHandler(__NR__GetGsDxDyOffset))
213 {
214 _GetGsDxDyOffset(gsGlobal->Mode, &gs_DX, &gs_DY, &gs_DW, &gs_DH);
215 gsGlobal->StartX += gs_DX;
216 gsGlobal->StartY += gs_DY;
217 }
218 }
219
220 // Keep the framebuffer in the center of the screen
221 gsGlobal->StartX += (gsGlobal->DW - ((gsGlobal->MagH + 1) * gsGlobal->Width )) / 2;
222 gsGlobal->StartY += (gsGlobal->DH - ((gsGlobal->MagV + 1) * gsGlobal->Height)) / 2;
223
224 if (gsGlobal->Interlace == GS_INTERLACED) {
225 // Do not change odd/even start position in interlaced mode
226 gsGlobal->StartY &= ~1;
227 }
228
229 // Calculate the actual display width and height
230 gsGlobal->DW = (gsGlobal->MagH + 1) * gsGlobal->Width;
231 gsGlobal->DH = (gsGlobal->MagV + 1) * gsGlobal->Height;
232
233 if ((gsGlobal->Interlace == GS_INTERLACED) && (gsGlobal->Field == GS_FRAME)) {
234 gsGlobal->MagV--;
235 }

238void gsKit_init_screen(GSGLOBAL *gsGlobal)
239{
240 u64 *p_data;
241 u64 *p_store;
242 int size = 19;
243
244 if((gsGlobal->Dithering == GS_SETTING_ON) &&
245 ((gsGlobal->PSM == GS_PSM_CT16) || (gsGlobal->PSM == GS_PSM_CT16S)))
246 size++;
247
248 gsKit_set_buffer_attributes(gsGlobal);
249
250#ifdef DEBUG
251 printf("Screen Mode:\n");
252 printf("------------\n");
253 printf("Width : %d\n", gsGlobal->Width);
254 printf("Height: %d\n", gsGlobal->Height);
255 printf("StartX: %d\n", gsGlobal->StartX);
256 printf("StartY: %d\n", gsGlobal->StartY);
257 printf("MagH : %d\n", gsGlobal->MagH);
258 printf("MagV : %d\n", gsGlobal->MagV);
259 printf("DW : %d\n", gsGlobal->DW);
260 printf("DH : %d\n", gsGlobal->DH);
261#endif
262
263 GS_RESET();
264
265 __asm__("sync.p; nop;");
266
267 *GS_CSR = 0x00000000; // Clean CSR registers
268
269 GsPutIMR(0x00007F00); // Masks all interrupts
270
271 SetGsCrt(gsGlobal->Interlace, gsGlobal->Mode, gsGlobal->Field);
272
273 // Fix 1080i frame mode
274 if ((gsGlobal->Mode == GS_MODE_DTV_1080I) && (gsGlobal->Field == GS_FRAME))
275 GS_SET_SMODE2(1, 1, 0);
276
277 gsGlobal->FirstFrame = GS_SETTING_ON;
278
279 if(gsGlobal->ZBuffering == GS_SETTING_OFF)
280 {
281 gsGlobal->Test->ZTE = GS_SETTING_ON;
282 gsGlobal->Test->ZTST = 1;
283 }
284
285 DIntr(); // disable interrupts
286
287 GS_SET_PMODE( 0, // Read Circuit 1
288 1, // Read Circuit 2
289 0, // Use ALP Register for Alpha Blending
290 1, // Alpha Value of Read Circuit 2 for Output Selection
291 0, // Blend Alpha with output of Read Circuit 2
292 0x80); // Alpha Value = 1.0
293
294 GS_SET_DISPFB1( 0, // Frame Buffer Base Pointer (Address/2048)
295 gsGlobal->Width / 64, // Buffer Width (Address/64)
296 gsGlobal->PSM, // Pixel Storage Format
297 0, // Upper Left X in Buffer
298 0);
299
300 GS_SET_DISPFB2( 0, // Frame Buffer Base Pointer (Address/2048)
301 gsGlobal->Width / 64, // Buffer Width (Address/64)
302 gsGlobal->PSM, // Pixel Storage Format
303 0, // Upper Left X in Buffer
304 0); // Upper Left Y in Buffer
305
306 GS_SET_DISPLAY1(
307 gsGlobal->StartX + gsGlobal->StartXOffset, // X position in the display area (in VCK unit
308 gsGlobal->StartY + gsGlobal->StartYOffset, // Y position in the display area (in Raster u
309 gsGlobal->MagH, // Horizontal Magnification
310 gsGlobal->MagV, // Vertical Magnification
311 gsGlobal->DW - 1, // Display area width
312 gsGlobal->DH - 1); // Display area height
313
314 GS_SET_DISPLAY2(
315 gsGlobal->StartX + gsGlobal->StartXOffset, // X position in the display area (in VCK units)
316 gsGlobal->StartY + gsGlobal->StartYOffset, // Y position in the display area (in Raster units)
317 gsGlobal->MagH, // Horizontal Magnification
318 gsGlobal->MagV, // Vertical Magnification
319 gsGlobal->DW - 1, // Display area width
320 gsGlobal->DH - 1); // Display area height
321
322 GS_SET_BGCOLOR( gsGlobal->BGColor->Red, // Red
323 gsGlobal->BGColor->Green, // Green
324 gsGlobal->BGColor->Blue); // Blue
325
326 EIntr(); //enable interrupts
327
328 gsGlobal->CurrentPointer = 0; // reset vram pointer too
329
330 // Context 1
331 gsGlobal->ScreenBuffer[0] = gsKit_vram_alloc( gsGlobal, gsKit_texture_size(gsGlobal->Width, gsGlobal->Height, gsGlobal->PSM), GSKIT_ALLOC_SYSBUFFER );
332
333 if(gsGlobal->DoubleBuffering == GS_SETTING_OFF)
334 {
335 gsGlobal->ScreenBuffer[1] = gsGlobal->ScreenBuffer[0];
336 }
337 else
338 // Context 2
339 gsGlobal->ScreenBuffer[1] = gsKit_vram_alloc( gsGlobal, gsKit_texture_size(gsGlobal->Width, gsGlobal->Height, gsGlobal->PSM), GSKIT_ALLOC_SYSBUFFER );
340
341
342 if(gsGlobal->ZBuffering == GS_SETTING_ON)
343 gsGlobal->ZBuffer = gsKit_vram_alloc( gsGlobal, gsKit_texture_size(gsGlobal->Width, gsGlobal->Height, gsGlobal->PSMZ), GSKIT_ALLOC_SYSBUFFER ); // Z Buffer
344
345 gsGlobal->TexturePointer = gsGlobal->CurrentPointer; // first useable address for textures
346
347 p_data = p_store = (u64 *)gsGlobal->dma_misc;
348
349 *p_data++ = GIF_TAG( size - 1, 1, 0, 0, GSKIT_GIF_FLG_PACKED, 1 );
350 *p_data++ = GIF_AD;
351
352 *p_data++ = 1;
353 *p_data++ = GS_PRMODECONT;
354
355 *p_data++ = GS_SETREG_FRAME_1( gsGlobal->ScreenBuffer[0] / 8192, gsGlobal->Width / 64, gsGlobal->PSM, 0 );
356 *p_data++ = GS_FRAME_1;
357
358 *p_data++ = GS_SETREG_XYOFFSET_1( gsGlobal->OffsetX,
359 gsGlobal->OffsetY);
360 *p_data++ = GS_XYOFFSET_1;
361
362 *p_data++ = GS_SETREG_SCISSOR_1( 0, gsGlobal->Width - 1, 0, gsGlobal->Height - 1 );
363 *p_data++ = GS_SCISSOR_1;
364
365 *p_data++ = GS_SETREG_TEST( gsGlobal->Test->ATE, gsGlobal->Test->ATST,
366 gsGlobal->Test->AREF, gsGlobal->Test->AFAIL,
367 gsGlobal->Test->DATE, gsGlobal->Test->DATM,
368 gsGlobal->Test->ZTE, gsGlobal->Test->ZTST );
369
370 *p_data++ = GS_TEST_1;
371
372 *p_data++ = GS_SETREG_CLAMP(gsGlobal->Clamp->WMS, gsGlobal->Clamp->WMT,
373 gsGlobal->Clamp->MINU, gsGlobal->Clamp->MAXU,
374 gsGlobal->Clamp->MINV, gsGlobal->Clamp->MAXV);
375
376 *p_data++ = GS_CLAMP_1;
377
378 if(gsGlobal->ZBuffering == GS_SETTING_ON)
379 {
380 if((gsGlobal->PSM == GS_PSM_CT16) && (gsGlobal->PSMZ != GS_PSMZ_16))
381 gsGlobal->PSMZ = GS_PSMZ_16; // seems only non-S 16-bit z depth works with this mode
382 if((gsGlobal->PSM != GS_PSM_CT16) && (gsGlobal->PSMZ == GS_PSMZ_16))
383 gsGlobal->PSMZ = GS_PSMZ_16S; // other depths don't seem to work with 16-bit non-S z depth
384
385 *p_data++ = GS_SETREG_ZBUF_1( gsGlobal->ZBuffer / 8192, gsGlobal->PSMZ, 0 );
386 *p_data++ = GS_ZBUF_1;
387 }
388 if(gsGlobal->ZBuffering == GS_SETTING_OFF)
389 {
390 *p_data++ = GS_SETREG_ZBUF_1( 0, gsGlobal->PSM, 1 );
391 *p_data++ = GS_ZBUF_1;
392 }
393
394 *p_data++ = GS_SETREG_COLCLAMP( 255 );
395 *p_data++ = GS_COLCLAMP;
396
397 *p_data++ = GS_SETREG_FRAME_1( gsGlobal->ScreenBuffer[1] / 8192, gsGlobal->Width / 64, gsGlobal->PSM, 0 );
398 *p_data++ = GS_FRAME_2;
399
400 *p_data++ = GS_SETREG_XYOFFSET_1( gsGlobal->OffsetX,
401 gsGlobal->OffsetY);
402 *p_data++ = GS_XYOFFSET_2;
403
404 *p_data++ = GS_SETREG_SCISSOR_1( 0, gsGlobal->Width - 1, 0, gsGlobal->Height - 1);
405 *p_data++ = GS_SCISSOR_2;
406
407 *p_data++ = GS_SETREG_TEST( gsGlobal->Test->ATE, gsGlobal->Test->ATST,
408 gsGlobal->Test->AREF, gsGlobal->Test->AFAIL,
409 gsGlobal->Test->DATE, gsGlobal->Test->DATM,
410 gsGlobal->Test->ZTE, gsGlobal->Test->ZTST );
411
412 *p_data++ = GS_TEST_2;
413
414 *p_data++ = GS_SETREG_CLAMP(gsGlobal->Clamp->WMS, gsGlobal->Clamp->WMT,
415 gsGlobal->Clamp->MINU, gsGlobal->Clamp->MAXU,
416 gsGlobal->Clamp->MINV, gsGlobal->Clamp->MAXV);
417
418 *p_data++ = GS_CLAMP_2;
419
420 if(gsGlobal->ZBuffering == GS_SETTING_ON)
421 {
422 *p_data++ = GS_SETREG_ZBUF_1( gsGlobal->ZBuffer / 8192, gsGlobal->PSMZ, 0 );
423 *p_data++ = GS_ZBUF_2;
424 }
425 if(gsGlobal->ZBuffering == GS_SETTING_OFF)
426 {
427 *p_data++ = GS_SETREG_ZBUF_1( 0, gsGlobal->PSM, 1 );
428 *p_data++ = GS_ZBUF_2;
429 }
430
431 *p_data++ = GS_BLEND_BACK2FRONT;
432 *p_data++ = GS_ALPHA_1;
433
434 *p_data++ = GS_BLEND_BACK2FRONT;
435 *p_data++ = GS_ALPHA_2;
436
437 *p_data++ = GS_SETREG_DIMX(gsGlobal->DitherMatrix[0],gsGlobal->DitherMatrix[1],
438 gsGlobal->DitherMatrix[2],gsGlobal->DitherMatrix[3],gsGlobal->DitherMatrix[4],
439 gsGlobal->DitherMatrix[5],gsGlobal->DitherMatrix[6],gsGlobal->DitherMatrix[7],
440 gsGlobal->DitherMatrix[8],gsGlobal->DitherMatrix[9],gsGlobal->DitherMatrix[10],
441 gsGlobal->DitherMatrix[11],gsGlobal->DitherMatrix[12],gsGlobal->DitherMatrix[13],
442 gsGlobal->DitherMatrix[14],gsGlobal->DitherMatrix[15]); // 4x4 dither matrix
443
444 *p_data++ = GS_DIMX;
445
446 *p_data++ = GS_SETREG_TEXA(0x00, 0, 0x80);
447 *p_data++ = GS_TEXA;
448
449 if((gsGlobal->Dithering == GS_SETTING_ON) && ((gsGlobal->PSM == GS_PSM_CT16) || (gsGlobal->PSM == GS_PSM_CT16S))) {
450 *p_data++ = 1;
451 *p_data++ = GS_DTHE;
452 }
453
454 dmaKit_send_ucab(DMA_CHANNEL_GIF, p_store, size);
455 dmaKit_wait_fast();
456
457 // do a single frame here to get rid whatever needs to be done during the first frame
458
459 gsKit_clear(gsGlobal, GS_SETREG_RGBAQ(0x00,0x00,0x00,0x00,0x00)); // clear the screen
460
461 gsKit_queue_exec(gsGlobal);
462
463 gsKit_sync_flip(gsGlobal);
464
465 gsKit_queue_reset(gsGlobal->Os_Queue);
466}
